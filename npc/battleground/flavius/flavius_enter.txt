// Flavius Officer - Guillaume
//============================================================
bat_room,133,178,5	script	Flavius Officer#01a	418,{
	if (checkweight(1201,1) == 0) {
		mes "- Espere um minuto!! -";
		mes "- - Atualmente você está carregando -";
		mes "- - demasiados itens com você. -";
		mes "- Por favor, tente novamente -";
		mes "- depois que você perder algum peso. -";
		close;
	}
	mes "[Guillaume Army Officer]";
	mes "Vamos mostrar o poder do Exército Guillaume para aqueles Croixs fedorentos!";
	next;
	switch(select("Eu quero me alistar em seu exército!:Fim da Conversa")) {
	case 1:
		if ((Class == Job_Novice) || (BaseClass == Job_Super_Novice)) {
			mes "[Guillaume Army Officer]";
			mes "Estou muito satisfeito que você queira se juntar a nós e lutar contra o Exército Croix, mas sinto muito: não posso enviar um novato como você para morrer no cruel campo de batalha.";
		}
		else {
			if (BaseLevel < 80) {
				mes "[Guillaume Army Officer]";
				mes "Estou muito satisfeito que você queira se juntar a nós e lutar contra o Exército Croix, mas sinto muito: não posso enviar um novato como você para morrer no cruel campo de batalha.";
			}
			else {
				set .@chk_urtime,checkquest(2070,PLAYTIME);
				if ((.@chk_urtime == 0)|| (.@chk_urtime == 1)) {
					mes "[Guillaume Army Officer]";
					mes "Você parece ter acabado de voltar do campo de batalha";
					mes "É muito cedo para você voltar. Vá descansar, e deixe os inimigos conosco";
				}
				else {
					if (.@chk_urtime == 2)
						erasequest 2070;
					set .@mapcount2,getmapusers("bat_b01");
					if (.@mapcount2 > 0) {
						mes "[Guillaume Army Officer]";
						mes "Recebi um relatório informando que um corpo de elite já foi despachado para o campo de batalha contra os Croixs";
						mes "Por que você não vai esperar um pouco? Sugiro que você afie suas armas e prepare seus suprimentos até lá";
					}
					else {
						if ($@FlaviusBG1 > 0) {
							mes "[Guillaume Army Officer]";
							mes "Um corpo de elite já está a postos para ser despachado para o campo de batalha";
							mes "Por que você não vai esperar um pouco? Sugiro que você afie suas armas e prepare seus suprimentos até lá";
						}
						else {
							mes "[Guillaume Army Officer]";
							mes "Você definitivamente parece estar pronto para a batalha";
							mes "Vá mostrar aos Croixs o que o medo realmente significa!";
							mes "Hoje, nosso grito de vitória deve ecoar por todo o campo de batalha!";
							close2;
							warp "bat_room",85,223;
							end;
						}
					}
				}
			}
		}
		break;
	case 2:
		mes "[Guillaume Army Officer]";
		mes "Hoje, seremos vitoriosos!";
		break;
	}
	close;
}
