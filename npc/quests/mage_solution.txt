pay_arche,122,100,0	script	Dollshoi	88,{
	mes "[Feiticeiro dos Magos]";
	mes "Você quer uma Solução de Payon? Hmm, me dê 50 Zenys e pelo menos 1 Tubo de Ensaio vazio.";
	next;
	if (select("Tudo bem, vamos negóciar.:Nah, esqueça.") == 1) {
		mes "[Feiticeiro dos Magos]";
		if (Zeny < 50) {
			mes "Ei! Você não tem dinheiro suficiente para cobrir minha taxa de 50 Zenys.";
			close;
		}
		if (countitem(1092) == 0) {
			mes "Você não pode adquirir a solução sem a garrafa! Traga-me um Tubo de Ensaio Vazio.";
			close;
		}
		delitem 1092,1; //Tubo de Ensaio Vazio
		set Zeny, Zeny-50;
		getitem 1089,1; //Solução de Payon
	}
	close;
}
