moc_para01,41,38,4	script	LvL 80	10281,{

	mes "[LvL 80]";
	mes "Olá aventureiro!";
	next;

	switch(select("Iniciar LvL 80","Entregar Recompensa","Cancelar")){
		case 1:
			callsub S_Start;
			end;
		case 2:
			callsub S_Complete;
			end;
		default:
			close;
	}
	end;

//==============================
// INICIAR QUEST
//==============================
S_Start:

	if (checkquest(60319) >= 0) {
		mes "[LvL 80]";
		mes "Você já iniciou esta missão!";
		close;
	}

	setquest 60319;
	mes "[LvL 80]";
	mes "Cace todos os dragões e volte quando terminar!";
	close;
	end;

//==============================
// ENTREGA
//==============================
S_Complete:

	if (checkquest(60319,HUNTING) == 2) {
		erasequest 60319;
		mes "[LvL 80]";
		mes "Excelente trabalho!";
		getexp 30000,20000;
		getitem Golden_Coin_,30;
		close;
	} else {
		mes "[LvL 80]";
		mes "Você ainda não derrotou todos os dragões.";
		close;
	}
	end;
}
