	mes "[Stephen]";

	next;
	mes "[Stephen]";
	mes "Chocolate!";
	mes "Isso mesmo, você não ama chocolate... eu amo.";
	mes "E você está com sorte, porque eu estou vendendo um pedaço por apenas 5.000 zeny!";
	next;
	menu "Eu quero um pouco de chocolate!",M_CHOCO,"Não, obrigado.",-;

	mes "[Stephen]";
	mes "Você não quer chocolate?";
	mes "Estou te dizendo! Você vai se arrepender!";
	mes "É melhor você pegar um pouco agora... você não vai encontrar esse tipo de Chocolate assim nunca mais!";
	mes "Pense bem sobre isso e me visite de novo algum dia.";
	close;

M_CHOCO:
	mes "[Stephen]";
	mes "Hah!";
	mes "Eu sabia disso!";
	mes "Mas eu não posso te vender mais do que 5 de uma vez... mas, se você realmente precisa de mais...";
	mes "você poderá me procurar novamente.";
	mes "Então quantos você quer?";
	next;
	set @needmon,0;
	input @flag_num;
	if (@flag_num <= 0) goto L_NONE;
	if (@flag_num > 5) goto L_ERR;
	set @needmon,@flag_num*5000;
	if (Zeny < @needmon) goto L_NOTENO;
	set Zeny,Zeny - @needmon;
	getitem 558,@flag_num;
	mes "[Stephen]";
	mes "Obrigado, pode ir agora!";
	mes "Você pode dar isso a alguém como um presente, ou aproveite você mesmo!";
	mes "Mmm.... chocolate doce...";
	mes "Visite-me a qualquer hora...!";
	close;

L_ERR:
	mes "[Stephen]";
	mes "Sinto muito, mas não posso te dar muitos.";
L_NONE:
	close;

L_NOTENO:
	mes "[Stephen]";
	mes "Sinto muito, mas parece que você não pode se dar ao luxo de comprar isso de mim.";
	close;
}
