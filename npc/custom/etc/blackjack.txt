//= Jogo de cartas Black Jack. Jogabilidade baseada em cassino padrão,
//  regras de black jack. O Negociante deve ter pelo menos 17 para ficar
//  e fica automaticamente em 17 e acima. O jogador deve ter pelo menos
//  13 para ficar. Ases contadas como 11 ou 1. Opção para "Dobrar".
//  Atualmente não permite "seguro" ou "divisão" de pares.                
//----- COMENTÁRIOS ADICIONAIS: ----------------------------------------------------

//= Cabeçalho e Tradução Pt-BR Por [ADM] Eragonat PK

cmd_in02,188,89,1	script	Negociante Black Jack	1_M_LIBRARYMASTER,{
	mes "[Negociante]";
	mes "Olá! Você gostaria de jogar o jogo Black Jack?";
M_Menu:
	next;
	menu "Quais são as regras?",M_0, "Eu quero jogar.",L_Play, "Não, obrigado.",M_End;

M_0:
	mes "[Negociante]";
	mes "O que você gostaria de saber?";
sM_Menu0:
	next;
	menu "O Básico.",sM_0a, "Ganhando e Perdendo.",sM_0b, "'Dobrar'.",sM_0c,
	"Valores do 'Ás'.",sM_0d, "Nada.",M_Menu;

sM_0a:
	mes "[Negociante]";
	mes "Black Jack é um jogo de cartas no qual o objetivo é chegar a 21 pontos, ou o";
	mes "mais perto possível de 21 pontos, sem ultrapassar 21.";
	next;
	mes "[Negociante]";
	mes "Antes do jogo começar, o jogador deve fazer uma aposta. Uma vez que a aposta";
	mes "foi feita, tanto o negociante quanto o jogador recebem 2 cartas";
	mes "cada um. Dependendo de quais cartas você tem, você poderá optar por ^5533FF'ficar'^000000,";
	mes "ou você poderá escolher ^5533FF'puxar'^000000.";
	next;
	mes "[Negociante]";
	mes "Quando você escolhe 'ficar', você está dizendo ao negociante que você não";
	mes "precisa de mais cartas. Isso permite que o negociante puxe se ele/ela";
	mes "quiser. Para ficar, você deve ter ^FF3355pelo menos 13 pontos^000000. O";
	mes "negociante só poderá ficar quando ele tiver ^FF335517 pontos ou mais^000000.";
	next;
	mes "[Negociante]";
	mes "Quando você escolhe 'puxar', você está dizendo ao negociante que você quer";
	mes "outra carta. Puxando mais cartas você poderá aumentar seus pontos";
	mes "totais. Desde que você tenha ^FF3355menos de 21^000000 pontos, você poderá puxar uma";
	mes "carta do baralho.";
	goto sM_Menu0;
sM_0b:
	mes "[Negociante]";
	mes "Existem 3 maneiras de ganhar e perder no Black Jack.";
	next;
	mes "[Negociante]";
	mes "1.) No final de uma rodada de Black Jack, se você tiver mais pontos";
	mes "do que o negociante você ganhará a rodada. Por outro lado, se você tiver";
	mes "menos pontos do que o negociante você perderá.";
	next;
	mes "[Negociante]";
	mes "2.) Se você puxar uma carta que faz o seu total ultrapassar 21 pontos, você";
	mes "irá automaticamente perder a rodada. Isso é chamado de ^5533FF'Bust'^000000 (Extrapolar), que quer dizer que o jogador extrapolou o total de 21 pontos. Se";
	mes "o negociante extrapolar então você ganhará a rodada.";
	next;
	mes "[Negociante]";
	mes "3.) Se você tem um total de 21 pontos com as 2 primeiras cartas, você";
	mes "irá automaticamente ganhar a rodada. Isso é chamado de ^5533FF'Black Jack'^000000,";
	mes "e acontece quando você recebe uma carta 'Ás' e um 'valor10'. Se o";
	mes "negociante recebe um Black Jack o jogador perderá automaticamente a rodada.";
	next;
	mes "[Negociante]";
	mes "4.) Além de ganhar e perder, você pode empatar com o negociante. Se";
	mes "você e o negociante têm o mesmo total de pontos no final de uma";
	mes "rodada, isso resultará em um empate sem vencedor e sem perda ou ganho";
	mes "em dinheiro.";
	next;
	mes "[Negociante]";
	mes "Isso é chamado de ^5533FF'Push'^000000 com o negociante. Isso também";
	mes "se aplica tanto a você quanto ao negociante tendo Black Jack ao mesmo";
	mes "tempo.";
	goto sM_Menu0;
sM_0c:
	mes "[Negociante]";
	mes "A opção 'Dobrar' permite que você dobre sua aposta atual,";
	mes "mas com a desvantagem de que você só poderá puxar uma";
	mes "carta adicional. Esta opção só está disponível no começo de";
	mes "cada rodada.";
	next;
	mes "[Negociante]";
	mes "Um exemplo de quando dobrar é útil, é quando";
	mes "suas primeiras 2 cartas dão um total de 11. Você tem uma boa";
	mes "chance de conseguir 21 ou 20 com a próxima carta que você puxar. Isso";
	mes "seria uma boa mão para dobrar.";
	goto sM_Menu0;
sM_0d:
	mes "[Negociante]";
	mes "A carta 'Ás' é uma carta única no jogo de Black Jack porque";
	mes "pode ter 2 valores. Um 'Ás' pode ser contado como 11 pontos,";
	mes "ou apenas 1 ponto. Por exemplo, se você tivesse um 'Ás' e um 4, isso";
	mes "daria 15 ou 5 pontos.";
	next;
	mes "[Negociante]";
	mes "Se você decidisse ficar, o 'Ás' seria automaticamente contado como";
	mes "11 pontos para dar 15 pontos no total.";
	next;
	mes "[Negociante]";
	mes "Se você tivesse decidido puxar e receber um 9, o 'Ás' iria";
	mes "automaticamente ser contado como 1 ponto para lhe dar um total de 14";
	mes "pontos. Se o 'Ás' fosse contado como 11 pontos, você teria um";
	mes "total de pontos acima de 21 e teria perdido.";
	next;
	mes "[Negociante]";
	mes "É por causa da flexibilidade que você tem com a carta 'Ás' que faz";
	mes "está carta ser a mais poderosa do jogo.";
	goto sM_Menu0;
M_End:
	mes "[Negociante]";
	mes "Sinta-se à vontade para voltar a qualquer momento.";
	close;

//================
L_Play:
	mes "[Negociante]";
	mes "Por favor, façam suas apostas...";
	next;
	menu "2z",M_1a, "10z",M_1b, "20z",M_1c, "100z",M_1d, "Muito rico para o meu sangue...",M_End;

M_1a:
	if(Zeny < 2) goto sL_NotEnuf;
	set @bet, 2;
	goto L_Cont0;
M_1b:
	if(Zeny < 10) goto sL_NotEnuf;
	set @bet, 10;
	goto L_Cont0;
M_1c:
	if(Zeny < 20) goto sL_NotEnuf;
	set @bet, 20;
	goto L_Cont0;
M_1d:
	if(Zeny < 100) goto sL_NotEnuf;
	set @bet, 100;
	goto L_Cont0;

sL_NotEnuf:
	mes "[Negociante]";
	mes "Sinto muito, mas você não tem zeny suficiente para fazer essa aposta.";
	close;

L_Cont0:
	mes "(as cartas estão sendo distribuídas)";
	next;
	deletearray $@card[0],13;
	set @dealerTurn, 0;
	set @numP, 0;
	set @numD, 0;
	set @pAce, 0;
	set @dAce, 0;
	callsub sF_GetCards, @numP, @playCard[@numP], @playCard$[@numP], @pAce;
	callsub sF_GetCards, @numP, @playCard[@numP], @playCard$[@numP], @pAce;
	callsub sF_GetCards, @numD, @dealCard[@numD], @dealCard$[@numD], @dAce;
	callsub sF_GetCards, @numD, @dealCard[@numD], @dealCard$[@numD], @dAce;

//==============
L_Start:
	callsub sF_GetTot;

	mes "- Aqui estão as cartas ^FF5533NEGOCIANTE^000000:";
	if (@numD==2) callsub sF_D2cards;
	if (@numD==3) callsub sF_D3cards;
	if (@numD==4) callsub sF_D4cards;
	if (@numD==5) callsub sF_D5cards;
	mes "    O NEGOCIANTE tem: ^FF5533"+@dealTot+"^000000.";
	mes " ";
	mes "- Aqui estão ^5533FFYOUR^000000 cartas:";
	if (@numP==2) callsub sF_P2cards;
	if (@numP==3) callsub sF_P3cards;
	if (@numP==4) callsub sF_P4cards;
	if (@numP==5) callsub sF_P5cards;
	if(@pAce != 1 || @playTot == 21) mes "    Você Tem: ^5533FF" +@playTot+ "^000000.";
	if(@pAce == 1 && @playTot != 21) mes "    Você Tem: ^5533FF" +@playTot+ "^000000, ou ^5533FF" +(@playTot-10)+ "^000000.";
	next;
	if(@playTot==21 && @dealTot==21) goto sL_Push;
	if(@numP==2 && @playTot == 21) goto sL_Win;
	if(@numD==2 && @dealTot == 21) goto sL_Lose;
	if(@playTot > 21) goto sL_Lose;
	if(@dealTot > 21) goto sL_Win;
	if(@numP==2 && @dealerTurn==0) menu "Peça-me(Puxar)",M_Hit, "Ficar",M_Stay, "Dobrar",M_Double;
	if(@dealerTurn == 0) menu "Peça-me(Puxar)",M_Hit, "Ficar",M_Stay;

M_Stay:
	mes "[Negociante]";
	if(@playTot < 13) goto sL_PlayToLow;
	if(@dealTot > 16) mes "O Negociante permanece.";
	if(@dealTot > 16 || @numD == 5) goto L_Check;
	mes "O Negociante vai puxar...";
	next;
	callsub sF_GetCards, @numD, @dealCard[@numD], @dealCard$[@numD], @dAce;
	set @dealerTurn, 1;
	goto L_Start;

sL_PlayToLow:
	mes "Sinto muito, mas você não tem um total alto o suficiente para ficar. Você deve puxar.";
	next;
	goto M_Hit;

M_Hit:
	if(@numP == 5) goto M_Stay;
	callsub sF_GetCards, @numP, @playCard[@numP], @playCard$[@numP], @pAce;
	goto L_Start;
M_Double:
	mes "[Negociante]";
	mes "O jogador escolheu Dobrar. Sua aposta atual será";
	mes "dobrada, e você só será capaz de puxar 1 carta extra.";
	next;
	set @dealerTurn, 1;
	set @bet, @bet*2;
	callsub sF_GetCards, @numP, @playCard[@numP], @playCard$[@numP], @pAce;
	goto L_Start;

//=============
L_Check:
	next;
	if(@playTot < @dealTot) goto sL_Lose;
	if(@playTot == @dealTot) goto sL_Push;

sL_Win:
	mes "[Negociante]";
	mes "Parabéns, você ganhou!";
	next;
	set Zeny, Zeny + @bet;
	goto L_Play;
sL_Lose:
	mes "[Negociante]";
	mes "Sinto muito, mas você perdeu.";
	set Zeny, Zeny - @bet;
	next;
	goto L_Play;
sL_Push:
	mes "[Negociante]";
	mes "É um empate. Você empatou com o Negociante.";
	next;
	goto L_Play;


//==================================
	// Sub função para dar as cartas / puxar as cartas
sF_GetCards:
	set @rnd, rand(1,13);
	if($@card[@rnd] == 4) goto sF_GetCards;
	set $@card[@rnd], $@card[@rnd] + 1;
	set getarg(1), @rnd;
	if(getarg(1) > 10) set getarg(1), 10;
	if(getarg(1) == 1 && getarg(3) < 1) set getarg(1), 11;
	if(getarg(1) == 11) set getarg(3), 1;
	set getarg(2), " " + getarg(1) + " ";
	if(@rnd == 10) set getarg(2), getarg(1);
	if(@rnd == 1) set getarg(2), " A ";
	if(@rnd == 11) set getarg(2), " J ";
	if(@rnd == 12) set getarg(2), " Q ";
	if(@rnd == 13) set getarg(2), " K ";
	set getarg(0), getarg(0) + 1;
	return;

//==================================
	// Sub função para obter a pontuação total para cada mão
sF_GetTot:
	set @i, 0;
	set @dealTot, 0;
	set @playTot, 0;

GetDealTot:
	set @dealTot, @dealTot + @dealCard[@i];
	set @i, @i + 1;
	if(@i < @numD) goto GetDealTot;
	if(@dAce == 1 && @dealTot > 21) set @dealTot, @dealTot - 10;
	set @i, 0;
GetPlayTot:
	set @playTot, @playTot + @playCard[@i];
	set @i, @i + 1;
	if(@i < @numP) goto GetPlayTot;
	if(@pAce == 1 && @playTot > 21) set @pAce, 2;
	if(@pAce > 0 && @playTot > 21) set @playTot, @playTot - 10;
	return;

//=======================================
	// Sub função para exibir as cartas
sF_D2cards:
	mes "    .-----.   .-----. ";
	mes "    | "+@dealCard$[0]+" |   | "+@dealCard$[1]+" | ";
	mes "    '-----'   '-----' ";
	return;
sF_P2cards:
	mes "    .-----.   .-----. ";
	mes "    | "+@playCard$[0]+" |   | "+@playCard$[1]+" | ";
	mes "    '-----'   '-----' ";
	return;
sF_D3cards:
	mes "    .-----.   .-----.   .-----. ";
	mes "    | "+@dealCard$[0]+" |   | "+@dealCard$[1]+" |   | "+@dealCard$[2]+" | ";
	mes "    '-----'   '-----'    '-----' ";
	return;
sF_P3cards:
	mes "    .-----.   .-----.   .-----. ";
	mes "    | "+@playCard$[0]+" |   | "+@playCard$[1]+" |   | "+@playCard$[2]+" | ";
	mes "    '-----'   '-----'    '-----' ";
	return;
sF_D4cards:
	mes " .-----.  .-----.  .-----. ";
	mes " | "+@dealCard$[0]+" |  | "+@dealCard$[1]+" |  | "+@dealCard$[2]+" | ";
	mes " '-----'  '-----'   '-----' ";
	mes "    .-----.  ";
	mes "    | "+@dealCard$[3]+" |";
	mes "    '-----'   ";
	return;
sF_P4cards:
	mes " .-----.  .-----.  .-----. ";
	mes " | "+@playCard$[0]+" |  | "+@playCard$[1]+" |  | "+@playCard$[2]+" |";
	mes " '-----'  '-----'   '-----' ";
	mes "    .-----.  ";
	mes "    | "+@playCard$[3]+" |";
	mes "    '-----'   ";
	return;
sF_D5cards:
	mes " .-----.  .-----.  .-----. ";
	mes " | "+@dealCard$[0]+" |  | "+@dealCard$[1]+" |  | "+@dealCard$[2]+" | ";
	mes " '-----'  '-----'   '-----' ";
	mes "    .-----.   .-----. ";
	mes "    | "+@dealCard$[3]+" |   | "+@dealCard$[4]+" | ";
	mes "    '-----'   '-----' ";
	return;
sF_P5cards:
	mes " .-----.  .-----.  .-----. ";
	mes " | "+@playCard$[0]+" |  | "+@playCard$[1]+" |  | "+@playCard$[2]+" | ";
	mes " '-----'  '-----'   '-----' ";
	mes "    .-----.   .-----. ";
	mes "    | "+@playCard$[3]+" |   | "+@playCard$[4]+" | ";
	mes "    '-----'   '-----' ";
	return;
}
