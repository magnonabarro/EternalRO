function SF_RetrieveRingM {
	mes "["+@name$+"]";
	if ($@wed_ring && countitem($@wed_ring) < 1) {
		mes "O que aconteceu com o seu "+getitemname($@wed_ring)+"? Você não perdeu... você? Precisamos continuar com a cerimônia!";
		close;
	}
	if ($wed_bride_sex)
		set @item, 2634;	//Anel do Noivo
	else
		set @item, 2635;	//Anel da Noiva
	if (getnameditem(@item,$wed_groom$) == 0) {
		mes "Você não parece ter espaço suficiente para carregar o anel... libere algum espaço e volte para receber a aliança do seu parceiro(a).";
		close;
	}
	mes "Aqui está o anel de casamento para sua noiva.";
	if ($@wed_ring) delitem $@wed_ring,1;
	set $wed_groom_progress,4;
	
	if ($wed_bride_progress == 4)
		SF_StartCeremony();
	else {
		mes "Uma vez que sua noiva receba o anel, a cerimônia começará.";
		close;
	}
}

//Subfunção: Verifica se o noivo/noiva ainda está usando suas coisas.
function SF_equip_check {
	if (Sex == SEX_MALE && getequipid(EQI_ARMOR) != 7170) {
		mes "["+@name$+"]";
		mes "Criança, o que você fez com o seu "+getitemname(7170)+"?";
		emotion ET_THINK;
		return 0;
	}
	if (Sex == SEX_FEMALE && getequipid(EQI_ARMOR) != 2338) {
		mes "["+@name$+"]";
		mes "Criança, você deveria usar um "+getitemname(2338)+" o tempo todo durante a cerimônia...";
		emotion ET_THINK;
		return 0;
	}
	if (Sex == SEX_FEMALE && $@wed_veil && getequipid(EQI_HEAD_TOP) != 2206) {
		mes "["+@name$+"]";
		mes "Criança, você não pode tirar o seu "+getitemname(2206)+" ainda....";
		emotion ET_THINK;
		return 0;
	}
	return 1;
}

function SF_Register {
	if ($@wed_allow) { //Seleciona função
		mes "["+@name$+"]";
		mes "Muito bem, você gostaria de se registrar como?";
		next;
		set @submenu, select("Noivo","Noiva","Cancelar");
	} else if (Sex == SEX_MALE) { //Noivo
		mes "["+@name$+"]";
		mes "Muito bem, você vai se registrar como o Noivo?";
		next;
		if (select("Sim","Eu mudei de idéia.")==1)
			set @submenu, 1;
		else
			set @submenu, 3;
	} else { //Noiva
		mes "["+@name$+"]";
		mes "Muito bem, você se registrará como a Noiva?";
		next;
		if (select("Sim","Eu mudei de idéia.")==1)
			set @submenu, 2;
		else
			set @submenu, 3;
	}
	switch (@submenu) {
	case 1: //Noivo
		SF_TryRegister(0);
		set $wed_progress,1;
		mes "["+@name$+"]";
		mes "Agora você está registrado como o noivo.";
		mes "Diga a sua noiva para se registrar o mais rápido possível.";
		emotion ET_SCRATCH;
		initnpctimer;
		close;
	case 2: //Noiva
		SF_TryRegister(1);
		set $wed_progress,1;
		mes "["+@name$+"]";
		mes "Você está registrado como a noiva.";
		mes "Diga ao seu noivo para se registrar o mais rápido possível.";
		emotion ET_SCRATCH;
		initnpctimer;
		close;
	default: //Cancelar
		mes "["+@name$+"]";
		mes "Volte quando estiver pronto.";
		close;
	}
}

//Subfunção SF_TryRegister (int bride)
function SF_TryRegister {
	set @bride, getarg(0);
	set @type$, "groom";
	if (@bride)
		set @type$, "bride";
		
	mes "["+@name$+"]";
	mes "Antes de se registrar como "+@type$+", deixe-me verificar se você atende a todos os requisitos...";
	next;
	if (Upper == 2) {
		mes "["+@name$+"]";
		mes "Oh querida(o), você é jovem demais para pensar em casamento!";
		emotion ET_SURPRISE;
		close;
	}
	if (Sex == SEX_MALE)
		set @item, 7170;
	else
		set @item, 2338;
	
	if (getequipid(EQI_ARMOR) != @item) {
		mes "["+@name$+"]";
		mes "Você deveria estar usando um "+getitemname(@item)+" se você quer se casar.";
		close;
	}
	if (Sex == SEX_FEMALE && $@wed_veil && getequipid(EQI_HEAD_TOP) != 2206) {
		mes "["+@name$+"]";
		mes "Onde está o seu "+getitemname(2206)+"? É um complemento necessário para o seu vestido.";
		close;
	}
	if ($@wed_ring && countitem($@wed_ring) < 1) {
		mes "["+@name$+"]";
		mes "Onde está o anel? Você precisa de um "+getitemname($@wed_ring)+" para poder trocar, querida..";
		close;
	}
	if (@bride)
		set @cost, $@wed_bride_reg;
	else
		set @cost, $@wed_groom_reg;

	if (Zeny < @cost) {
		mes "["+@name$+"]";
		mes "Sinto muito, mas você não tem zeny o suficiente para pagar a taxa de inscrição.";
		mes "Volte depois de coletar "+@cost+" zeny(s).";
		close;
	}
	set Zeny,Zeny-@cost;
	sc_start SC_Wedding,3600000,1; //Inicia Efeito Casamento (SC_WEDDING)
	if (@bride) {
		set $wed_bride_progress,1;
		set $wed_bride$,strcharinfo(0);
		set $wed_bride_sex, Sex;
		if ($@wedding_effect == 1) //Armazena o id da conta para efeito.
			set $wedding_effect_id, getcharid(3);
	} else {
		set $wed_groom_progress,1;
		set $wed_groom$,strcharinfo(0);
		set $wed_groom_sex, Sex;
		if ($@wedding_effect == 2) //Armazena o ID da conta para efeito.
			set $wedding_effect_id, getcharid(3);
	}
}
