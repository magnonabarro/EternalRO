function	script	SC_DISPARM	{

	if(@imun > 0){ emotion ET_KIK; dispbottom "Armadilha falhou! Você está imune."; return; }

	ARMEXIT:

	set @SC_RANDOM,rand(1,100);

	if(@SC_RANDOM <= $SC_ARMEXIT) {
	specialeffect2 69;
	specialeffect2 304;
	dispbottom "Armadilha! Você será jogado para fora";
	warp "prontera",156,185;
	emotion ET_CRY;
	return;
	} else {
	goto ARMDEG;
	}
	
ARMDEG:

	set @SC_RANDOM,rand(1,100);

	if(@SC_RANDOM <= $SC_ARMDEG) {
	specialeffect2 38;
	specialeffect2 234;
	specialeffect2 372;
	emotion ET_FRET;
	dispbottom "Armadilha! Velocidade Diminuida!";
	sc_start SC_DECREASEAGI,80000,50;
	sc_start SC_CHANGEUNDEAD,50000,0;
	return;
	} else {
	goto ARMDISPELL;
	}

ARMDISPELL:

	set @SC_RANDOM,rand(1,100);

	if(@SC_RANDOM <= $SC_ARMDISPELL) {
	specialeffect2 235;
	emotion ET_HUK;
	dispbottom "Armadilha! Todos efeitos foram retirados!";
	sc_end SC_ALL;
	return;
	} else {
	goto ARMIMUN;
	}

ARMIMUN:

	set @SC_RANDOM,rand(1,100);

	if(@SC_RANDOM <= $SC_ARMIMUN) {
	specialeffect2 675;
	specialeffect2 749;
	emotion ET_SMILE;
	dispbottom "Bônus! Você está imune as armadilhas por 15 segundos!";
	set @imun,1;
	sleep2 15000;
	set @imun,0;
	return;
	} else {
	goto ARMSTRIP;
	}

ARMSTRIP:

	set @SC_RANDOM,rand(1,100);

	if(@SC_RANDOM <= $SC_ARMSTRIP) {
	switch(rand(1,4)){
	case 1: if(getequipid(EQI_HAND_R)){ specialeffect2 269; dispbottom "Armadilha! "+getequipname(EQI_HAND_R)+" retirada!"; unequip EQI_HAND_R; } break;
	case 2: if(getequipid(EQI_HAND_R)){ specialeffect2 270; dispbottom "Armadilha! "+getequipname(EQI_HAND_R)+" retirado!"; unequip EQI_HAND_L; } break;
	case 3: if(getequipid(EQI_ARMOR)){ specialeffect2 271; dispbottom "Armadilha! "+getequipname(EQI_ARMOR)+" retirada!"; unequip EQI_ARMOR; } break;
	case 4: if(getequipid(EQI_HEAD_TOP)){ specialeffect2 272; dispbottom "Armadilha! "+getequipname(EQI_HEAD_TOP)+" retirado!"; unequip EQI_HEAD_TOP; } break;
	}
	emotion ET_HUK;
	return;
	} else {
	goto ARMBON1;
	}

ARMBON1:

	set @SC_RANDOM,rand(1,100);

	if(@SC_RANDOM <= $SC_ARMBON1) {
	specialeffect2 100;
	specialeffect2 847;
	emotion ET_KIK;
	dispbottom "Bônus! Você recebeu 2 Bandeiras Nacionais!";
	getitem 7549,2;
	return;
	} else {
	goto ARMBON2;
	}

ARMBON2:

	set @SC_RANDOM,rand(1,100);

	if(@SC_RANDOM <= $SC_ARMBON2) {
	specialeffect2 100;
	emotion ET_SMILE;
	dispbottom "Bônus! Você recebeu 1 Bandeira Nacional!";
	getitem 7549,1;
	return;
	} else {
	goto ARMBLIND;
	}

ARMBLIND:

	set @SC_RANDOM,rand(1,100);

	if(@SC_RANDOM <= $SC_ARMBLIND) {
	if (checkoption(0x10)){ set @falcon, 1; }
	if (checkoption(0x20)){ set @peco, 1; }
	specialeffect2 108;
	emotion ET_KEK;
	dispbottom "Armadilha! Congelado!";
	atcommand "@option 2";
	if (@falcon > 0){ setoption 0x10; }
	if (@peco > 0){ setoption 0x20; }
	sleep2 6000;
	atcommand "@option 0";
	if (@falcon > 0){ setoption 0x10; set @falcon, 0; }
	if (@peco > 0){ setoption 0x20; set @peco, 0; }
	return;
	} else {
	goto ARMAGIL;
	}

ARMAGIL:

	set @SC_RANDOM,rand(1,100);

	if(@SC_RANDOM <= $SC_ARMAGIL) {
	dispbottom "Bônus! Você recebeu Aumentar Agilidade Nv. 10!";
	emotion ET_HUNGRY;
	sc_start 32,240000,10;
	skilleffect 29,0;
	specialeffect2 507;
	specialeffect2 519;
	return;
	} else {
	goto ARMCRIS;
	}

ARMCRIS:

	set @SC_RANDOM,rand(1,100);

	if(@SC_RANDOM <= $SC_ARMCRIS) {
	emotion ET_MONEY;
	pcblockmove getcharid(3),1;
	specialeffect2 60;
	specialeffect2 501;
	sleep2 2400;
	pcblockmove getcharid(3),0;
	dispbottom "Bônus! Teleporte para a Arena Premiada!";
	specialeffect2 227;
	warp "job_monk",0,0;
	return;
	} else {
	goto ARMBONN;
	}


ARMBONN:

	set @SC_RANDOM,rand(1,20);
	if(@SC_RANDOM <= 1 && countitem(7549) > 2){
	specialeffect2 722;
	specialeffect2 380;
	emotion ET_CRY;
	dispbottom "Armadilha! Você perdeu 3 Bandeiras Nacionais!";
	delitem 7549,3;
	return;
	}else{ goto ARMFIM; }

ARMFIM:

	set @SC_RANDOM,rand(1,10);
	if(@SC_RANDOM <= 5 && countitem(7549) > 0){
	specialeffect2 18;
	emotion ET_KEK;
	dispbottom "Armadilha! Você perdeu 1 Bandeira Nacional!";
	delitem 7549,1;
	return;
	}
}
