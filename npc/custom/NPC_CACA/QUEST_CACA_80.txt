sala_caca,3,7,5	script	Caça LvL 80#NPC	49,{

	mes "[Caça LvL 80]";
	mes "Olá aventureiro!";
	next;

	switch(select("Iniciar Caça LvL 80","Entregar Recompensa","Cancelar")){
		case 1:
			callsub S_Start;
			end;
		case 2:
			callsub S_Complete;
			end;
		default:
			close;
	}
	end;

//==============================
// INICIAR QUEST
//==============================
S_Start:

	if (checkquest(60315) >= 0) {
		mes "[Caça LvL 80]";
		mes "Você já iniciou esta missão!";
		close;
	}

	setquest 60315;
	mes "[Caça LvL 80]";
	mes "Cace todos os dragões e volte quando terminar!";
	close;
	end;

//==============================
// ENTREGA
//==============================
S_Complete:

	if (checkquest(60315,HUNTING) == 2) {
		erasequest 60315;
		mes "[Caça LvL 80]";
		mes "Excelente trabalho!";
		getexp 30000,20000;
		getitem Silver_Coin,50;
		close;
	} else {
		mes "[Caça LvL 80]";
		mes "Você ainda não derrotou todos os dragões.";
		close;
	}
	end;
}
