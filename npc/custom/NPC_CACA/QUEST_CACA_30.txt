sala_caca,16,7,4	script	Caça LvL 30#NPC	52,{

	mes "[Caça LvL 30]";
	mes "Olá aventureiro!";
	next;

	switch(select("Iniciar Caça LvL 30","Entregar Recompensa","Cancelar")){
		case 1:
			callsub S_Start;
			end;
		case 2:
			callsub S_Complete;
			end;
		default:
			close;
	}
	end;

//==============================
// INICIAR QUEST
//==============================
S_Start:

	if (checkquest(60312) >= 0) {
		mes "[Caça LvL 30]";
		mes "Você já iniciou esta missão!";
		close;
	}

	setquest 60312;
	mes "[Caça LvL 30]";
	mes "Cace todos os dragões e volte quando terminar!";
	close;
	end;

//==============================
// ENTREGA
//==============================
S_Complete:

	if (checkquest(60312,HUNTING) == 2) {
		erasequest 60312;
		mes "[Caça LvL 30]";
		mes "Excelente trabalho!";
		getexp 30000,20000;
		getitem Silver_Coin,20;
		close;
	} else {
		mes "[Caça LvL 30]";
		mes "Você ainda não derrotou todos os dragões.";
		close;
	}
	end;
}
