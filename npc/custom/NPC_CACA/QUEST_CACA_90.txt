sala_caca,3,12,5	script	Caça LvL 90#NPC	48,{

	mes "[Caça LvL 90]";
	mes "Olá aventureiro!";
	next;

	switch(select("Iniciar Caça LvL 90","Entregar Recompensa","Cancelar")){
		case 1:
			callsub S_Start;
			end;
		case 2:
			callsub S_Complete;
			end;
		default:
			close;
	}
	end;

//==============================
// INICIAR QUEST
//==============================
S_Start:

	if (checkquest(60316) >= 0) {
		mes "[Caça LvL 90]";
		mes "Você já iniciou esta missão!";
		close;
	}

	setquest 60316;
	mes "[Caça LvL 90]";
	mes "Cace todos os dragões e volte quando terminar!";
	close;
	end;

//==============================
// ENTREGA
//==============================
S_Complete:

	if (checkquest(60316,HUNTING) == 2) {
		erasequest 60316;
		mes "[Caça LvL 90]";
		mes "Excelente trabalho!";
		getexp 30000,20000;
		getitem Silver_Coin,60;
		close;
	} else {
		mes "[Caça LvL 90]";
		mes "Você ainda não derrotou todos os dragões.";
		close;
	}
	end;
}
