// callfunc("F_GM_NPC"{,<senha>,<0:int | 1:for>{,<mín>,<máx>}})
function	script	F_GM_NPC	{

	// Para desabilitar todos os NPCs GM (Administração), descomente a linha abaixo.
	//end;

	// Para definir um nível de GM mínimo para acessar os NPCs, edite a linha abaixo.
	if (getgroupid() < 99) {
		// Log de evento.
		getmapxy(.@map$,.@x,.@y,BL_NPC);
		logmes strcharinfo(0)+" tentou acessar os NPCs de Administração "+strnpcinfo(0)+" ("+.@map$+","+.@x+","+.@y+").";
		end;
	}

	if (!getargcount()) return;

	// Para desativar as senhas, descomente a linha abaixo.
	//return 1;

	// Valores de retorno:
	//  1: Senha correta
	//  0: Senha incorreta
	// -1: Zero/cancelado (somente inteiro)
	// -2: Fora do limite (somente inteiro)
	if (getarg(1) == 0) {
		// Senha Numérica
		if (getarg(3,0)) {
			.@error = input(.@input,getarg(2),getarg(3));
			if (.@error)
				return -2;
		} else
			input .@input;
		if (.@input == 0)
			return -1;
		else
			return (.@input == getarg(0))?1:0;
	} else {
		// Senha da String
		input .@inputstr$;
		return (.@inputstr$ == getarg(0))?1:0;
	}
}
