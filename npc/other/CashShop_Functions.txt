// Neuralizador
//======================================================================================
// - Reseta os Pontos de Habilidade. Mas para isto você não pode estar com items ou Carrinho/Falcão/Peco-Peco.
// - Nenhum argumento.
function	script	F_CashReset	{
	if (Class != Job_Novice && Weight == 0 && !checkriding() && !checkfalcon() && !checkcart() && !ismounting() && !checkdragon()) {
		resetskill;
		sc_end SC_ALL;
		if (countitem(12213)) delitem 12213,1;
	}
	return;
}

// Poção de Redução de Atributos
//============================================================
// - Reduz permanentemente os atibutos base <tipo> em <val>.
// - Retorna pontos de Atributos iguais aos pontos necessários para elevar
//   esse Atributo ao valor original.
// - Não funciona se o atributo base <tipo> se tornar menor que 1 após a redução.
// * callfunc("F_CashReduceStat",<tipo>{,<val>,<ID_do_item>});
function	script	F_CashReduceStat	{
	.@type = getarg(0);
	.@amount = getarg(1, -1);
	.@itemid = getarg(2, 0);
	
	if((readparam(.@type) + .@amount) < 1) return;
	
	if(.@itemid) {
		if(countitem(.@itemid))
			delitem .@itemid,1;
		else
			return;
	}
	StatusPoint += needed_status_point(.@type, .@amount);
	statusup2 .@type,.@amount;
	return;
}

// Poção de Redução de Atributos de Características
//============================================================
// - Redução de Atributos de Característica p
// - Reduz permanentemente o Atributo de Característica base <tipo> em <val>.
// - Retorna pontos de Atributos de Característica iguais aos pontos necessários para elevar 
//   esse Atributo ao valor original.
// - Não funciona se o Atributo de Característica base <tipo> se tornar menor que 0 após a redução.
function	script	F_CashReduceTraitStat	{
	.@type = getarg(0);
	.@amount = getarg(1, -1);
	.@itemid = getarg(2, 0);
	
	if((readparam(.@type) + .@amount) < 0) return;
	
	if(.@itemid) {
		if(countitem(.@itemid))
			delitem .@itemid,1;
		else
			return;
	}
	TraitPoint += needed_trait_point(.@type, .@amount);
	traitstatusup2 .@type,.@amount;
	return;
}
