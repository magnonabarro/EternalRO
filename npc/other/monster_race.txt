p_track01,58,0,0	script	Tempo de Progresso da Corrida	-1,{
OnEnable:
	initnpctimer;
	enablenpc "Tempo de Progresso da Corrida";
	for ( .@c = 1; .@c < 7; .@c++ ) {
		.@line = rand(1,70);
		enablenpc "iniciando#"+.@c;
		enablenpc "Pneu"+.@c+"#1";
		if ( .@line <= 10 ) {
			.@tired = rand(50,60);
			enablenpc "Sorte"+.@c+"#5";
			enablenpc "Sorte"+.@c+"#6";
		} else if ( .@line <= 30 ) {
			.@tired = rand(40,60);
			enablenpc "Sorte"+.@c+"#5";
			if (.@tired >= 50) enablenpc "Pneu"+.@c+"#2";
		} else if ( .@line <= 40 ) {
			.@tired = rand(30,50);
			enablenpc "Sorte"+.@c+"#1";
			enablenpc "Pneu"+.@c+"#2";
			if ( .@tired < 40 ) enablenpc "Pneu"+.@c+"#3";
		} else if (.@line <= 50) {
			.@tired = rand(20,40);
			enablenpc "Sorte"+.@c+"#1";
			enablenpc "Sorte"+.@c+"#2";
			enablenpc "Pneu"+.@c+"#2";
			enablenpc "Pneu"+.@c+"#3";
			if (.@tired < 30) enablenpc "Pneu"+.@c+"#4";
		} else if (.@line <= 60) {
			.@tired = rand(10,30);
			for ( .@i = 1; .@i <= 3; .@i++ )
				enablenpc "Sorte"+.@c+"#"+.@i;
			for ( .@i = 2; .@i <= 4; .@i++ )
				enablenpc "Pneu"+.@c+"#"+.@i;
			if ( .@tired < 20) enablenpc "Pneu"+.@c+"#5";
		} else if (.@line <= 70) {
			.@tired = rand(0,20);
			for ( .@i = 1; .@i <= 4; .@i++ )
				enablenpc "Sorte"+.@c+"#"+.@i;
			for ( .@i = 2; .@i <= 5; .@i++ )
				enablenpc "Pneu"+.@c+"#"+.@i;
			if (.@tired < 10) enablenpc "Pneu"+.@c+"#6";
		}
		setd "$@mr_1_Sorte"+.@c, .@line;
		setd "$@mr_1_Pneu"+.@c, .@tired;
	}
	end;
OnTimer1000:
	enablenpc "Ajudante de Bilheteria#single";
	end;
OnTimer7000:
	mapannounce "p_track01","Bem-vindo à Arena de Corrida de Monstros.",bc_map,"0xffb6c1";
	end;
OnTimer10000:
	mapannounce "p_track01","Sinta-se à vontade para consultar o suporte técnico sempre que tiver dúvidas.",bc_map,"0xffb6c1";
	end;
OnTimer120000:
	mapannounce "p_track01","A Corrida de Monstro Simples começará em 3 minutos.",bc_map,"0xffb6c1";
	end;
OnTimer123000:
	mapannounce "p_track01","Por favor, fale com um Ajudante de Bilheteria se deseja apostar na corrida.",bc_map,"0xffb6c1";
	end;
OnTimer240000:
	mapannounce "p_track01","A Corrida de Monstros Simples começará em breve.",bc_map,"0xffb6c1";
	end;
OnTimer243000:
	mapannounce "p_track01","Por favor, pergunte a um Ajudante de Bilheteria se deseja apostar na corrida.",bc_map,"0xffb6c1";
	end;
OnTimer300000:
	mapannounce "p_track01","A Corrida dos Monstros já começou. Boa sorte a todos os participantes.",bc_map,"0xffb6c1";
	$@monster_race = 0;
	disablenpc "Ajudante de Bilheteria#single";
	for ( .@i = 1; .@i <= 6; .@i++ )
		donpcevent "Corredor Nº. "+.@i+"#"+.@i+"::OnEnable";
	stopnpctimer;
	end;

OnDisable:
OnInit:
	disablenpc "Tempo de Progresso da Corrida";
	end;
}
