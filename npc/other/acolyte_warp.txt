prontera,98,121,4	script	Noviça#Prtclear	90,{
	mes "[Keiki]";
	mes "Olá, aventureiro.";
	mes "Eu tenho estudado a magia de todo o reino de Rune-Midgard para atualizar o que eu acredito ser uma das maiores habilidades disponíveis para a classe Noviço.";
	next;

	if (is_function("VIP_iRO_Acolyte"))
		callfunc "VIP_iRO_Acolyte","[Keiki]","Juno";

	mes "[Keiki]";
	mes "Eu sou a única noviça que alcançou a habilidade do Portal Nível 10!";
	emotion ET_THINK, playerattached();
	next;
	mes "[Keiki]";
	mes "Isso mesmo! E...";
	mes "Eu te juro que não esqueço os lugares que eu já estive e memorizei.";
	mes "Um dia eu vou aumentar minhas habilidades para me teleportar para onde eu quiser~";
	next;
	mes "[Keiki]";
	mes "Estou disposta a te teleportar para os muitos locais que eu já memorizei por uma pequena taxa.";
	mes "Você gostaria de usar este serviço?";
	next;
	switch(select("Sim.:Não.")) {
	case 1:
		mes "[Keiki]";
		mes "Onde você gostaria de ir?";
		mes "Desejo-lhe boa sorte em sua jornada.";
		next;

		setarray .@towns$[0], "Izlude", "Geffen", "Payon", "Morroc", "Alberta", "Al De Baran", "Comodo", "Umbala", "Juno";
		setarray .@cost[0],        600,     1200,    1200,     1200,      1800,          2200,     2200,     2200,   1800;

		set .@size, getarraysize(.@towns$);
		for(set .@i,0; .@i<.@size; set .@i,.@i+1)
			set .@menu$, .@menu$+.@towns$[.@i]+" -> "+.@cost[.@i]+"z:";
		set .@i, select(.@menu$+"Cancelar")-1;
		if (.@i == .@size)
			close;
		if (Zeny < .@cost[.@i]) {
			mes "[Keiki]";
			mes "Sinto muito, mas você não tem";
			mes "zeny suficiente para o serviço de";
			mes "Teleporte. A taxa para se teletransportar";
			mes "para "+.@towns$[.@i]+" é de "+.@cost[.@i]+" zeny's.";
			close;
		}
		specialeffect2 EF_READYPORTAL;
		specialeffect2 EF_TELEPORTATION;
		specialeffect2 EF_PORTAL;
		next;
		set Zeny, Zeny - .@cost[.@i];
		switch(.@i) {
			case 0: if (checkre(0)) warp "izlude",128,98; else warp "izlude",91,105; break;
			case 1: warp "geffen",120,39; break;
			case 2: warp "payon",161,58; break;
			case 3: warp "morocc",156,46; break;
			case 4: warp "alberta",117,56; break;
			case 5: warp "aldebaran",168,112; break;
			case 6: warp "comodo",209,143; break;
			case 7: warp "umbala",100,154; break;
			case 8: warp "yuno",158,125; break;
		}
		close;
	case 2:
		close;
	}
}
