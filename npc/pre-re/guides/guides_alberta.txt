alberta,23,238,4	script	Guia#alb::AlbGuide	105,{
	cutin "prt_soldier",2;
	mes "[Guia de Alberta]";
	mes "Bem-vindo à Alberta,";
	mes "a cidade portuária. Pode me";
	mes "falar se você estiver tendo";
	mes "dificuldades para encontrar algo na";
	mes "cidade, ou se você só precisar de uma";
	mes "orientação sobre a cidade.";
	set .@loop1,1;
	while (.@loop1) {
		next;
		switch(select("Guia da Cidade:Remover Marcações do Mini-mapa:Informações:Cancelar")) {
		case 1:
			mes "[Guia de Alberta]";
			mes "Por favor, selecione";
			mes "um local do";
			mes "menu a seguir.";
			if (.@compass_check == 0) {
				mes "Você quer que eu";
				mes "marque os locais";
				mes "no seu Mini-mapa?";
				next;
				if (select("Sim.:Não.") == 1) set .@compass_check,1;
			}
			set .@loop2,1;
			while (.@loop2) {
				if (.@wait_button_chk == 0) set .@wait_button_chk,1;
				else next;
				
				switch(select("^FF0000Guilda dos Mercadores^000000:Loja de Armas:Loja de Utilidades.:Estalagem:Forja:Cancelar")) {
				case 1:
					mes "[Guia de Alberta]";
					mes "A Guilda dos Mercadores";
					mes "lida com mudanças de classe";
					mes "para a classe Mercador,";
					mes "e está localizada no";
					mes "canto sudoeste";
					mes "de Alberta.";
					if (.@compass_check)
						viewpoint 1,33,41,2,0xFF0000;
					break;
				case 2:
					mes "[Guia de Alberta]";
					mes "A Loja de Armas";
					mes "se encontra no";
					mes "extremo sul de Alberta.";
					if (.@compass_check)
						viewpoint 1,117,37,3,0xFF00FF;
					break;
				case 3:
					mes "[Guia de Alberta]";
					mes "A Loja de Utilidades";
					mes "é meio perto";
					mes "do centro de";
					mes "Alberta. Não deveria";
					mes "ser muito difícil de encontrar.";
					if (.@compass_check)
						viewpoint 1,98,154,4,0xFF00FF;
					break;
				case 4:
					mes "[Guia de Alberta]";
					mes "Há uma Estalagem";
					mes "no extremo norte";
					mes "de Alberta, onde";
					mes "você pode descansar.";
					if (.@compass_check)
						viewpoint 1,65,233,5,0xFF00FF;
					break;
				case 5:
					mes "[Guia de Alberta]";
					mes "A Forja em Alberta";
					mes "está no mesmo prédio";
					mes "que a Guilda dos Mercadores.";
					mes "Vá para o sudoeste.";
					if (.@compass_check)
						viewpoint 1,35,41,6,0xFF00FF;
					break;
				case 6:
					mes "[Guia de Alberta]";
					mes "Por favor, peça-me para";
					mes "''Remover Marcações do Mini-mapa''";
					mes "se você não quiser mais que as";
					mes "marcas de localizações sejam exibidas";
					mes "no seu Mini-mapa.";
					set .@loop2,0;
				}
			}
			break;
		case 2:
			viewpoint 2,237,41,2,0xFF0000;
			viewpoint 2,237,41,3,0xFF00FF;
			viewpoint 2,46,345,4,0xFF00FF;
			viewpoint 2,175,220,5,0xFF00FF;
			viewpoint 2,175,220,6,0xFF00FF;
			set .@compass_check,0;
			break;
		case 3:
			mes "[Guia de Alberta]";
			mes "Avanços na feitiçaria e";
			mes "tecnologia nos permitiram";
			mes "atualizar nossos sistemas de";
			mes "informações, permitindo marcar";
			mes "locais no seu Mini-mapa";
			mes "para facilitar a navegação.";
			next;
			mes "[Guia de Alberta]";
			mes "Seu Mini-mapa está localizado";
			mes "no canto superior direito";
			mes "da tela. Se você não puder";
			mes "vê-lo, pressione as teclas Ctrl + Tab";
			mes "ou clique no botão ''Mapa-múndi'' na sua Janela de Informações Básicas.";
			next;
			mes "[Guia de Alberta]";
			mes "No seu Mini-mapa,";
			mes "clique nos símbolos de ''+'' e ''-''";
			mes "para aumentar ou diminuir o";
			mes "Zoom do seu Mini-mapa.";
			mes "Esperamos que você aproveite suas";
			mes "viagens aqui na cidade de Alberta.";
			break;
		case 4:
			mes "[Guia de Alberta]";
			mes "Esteja seguro na suas";
			mes "viagens e não hesite";
			mes "em me perguntar se você tiver";
			mes "alguma dúvida sobre Alberta.";
			close2;
			set .@loop1,0;
		}
	}
	cutin "prt_soldier",255;
	end;
}
