veins,210,345,5	script	Guia de Veins#1::ve_guide	934,{
	mes "[Guia de Veins]";
	mes "A Cidade desértica de Veins dar boas vindas a os aventureiros que buscam abrigo contra as duras tempestades de areia.";
	mes "Se esta é a primeira vez que você usa os serviços de guia, por que você não checa primeiro...";
	set .@loop1,1;
	while(.@loop1) {
		next;
		switch(select("Guias da Cidade.:Remover Marcações do Mini-mapa.:Informações.:Cancelar!")) {
		case 1:
			mes "[Guia de Veins]";
			mes "Eu posso te dizer onde fica cada edifício que procura, desde que seja em Veins.";
			mes "Então, aonde você quer ir?";
			if (.@compass_check == 0) {
				mes "Você gostaria de que eu";
				mes "marque as indicações";
				mes "no seu Mini-mapa?";
				next;
				if (select("Sim.:Não.") == 1) set .@compass_check,1;
			}
			set .@loop2,1;
			while(.@loop2) {
				if (.@wait_button_chk == 0) set .@wait_button_chk,1;
				else next;

				switch(select("Templo.:Estalagem.:Loja de Armas.:Loja de Utilidades.:Aeroplano.:Taverna.:Instituto de Pesquisas Geológicas.:Cancelar!")) {
				case 1:
					mes "[Guia de Veins]";
					mes "Nosso Templo está localizado ao norte";
					mes "e é sempre cheio de seguidores fiéis da Deusa Freya.";
					if (.@compass_check == 1) viewpoint 1,196,258,1,0xFF0000;
					break;
				case 2:
					mes "[Guia de Veins]";
					mes "Você pode descansar o infado da jornada na Hospedaria.";
					mes "O prédio da esquerda ao meu lado é a Estalagem de Veins.";
					if (.@compass_check == 1) viewpoint 1,128,266,2,0xFF00FF;
					break;
				case 3:
					mes "[Guia de Veins]";
					mes "Sim, você deve se proteger do perigo sozinho.";
					mes "Compre armas de alta qualidade a preços acessíveis.";
					mes "A Loja de Armas de Veins está localizada a oeste.";
					if (.@compass_check == 1) viewpoint 1,150,175,3,0x99FFFF;
					break;
				case 4:
					mes "[Guia de Veins]";
					mes "Você já tem suprimentos suficientes para a sua aventura?";
					mes "Se não tem, sugiro que você veja o que a Loja de Utilidades de Veins, no centro, tem a oferecer.";
					if (.@compass_check == 1) viewpoint 1,230,161,4,0x0000FF;
					break;
				case 5:
					mes "[Guia de Veins]";
					mes "Por favor, esteja ciente de que Veins só opera aeronaves de carga.";
					if (.@compass_check == 1) viewpoint 1,273,285,5,0x00FF00;
					break;
				case 6:
					mes "[Guia de Veins]";
					mes "Se você gostaria de fazer amizade com";
					mes "as pessoas da cidade, eu sugiro que você";
					mes "vá tomar uma bebida na Taverna que está localizada";
					mes "na parte oeste da cidade.";
					if (.@compass_check == 1) viewpoint 1,150,217,6,0x00FF00;
					break;
				case 7:
					mes "[Guia de Veins]";
					mes "Você está interessado em estudar geologia?";
					mes "Então é melhor você checar o";
					mes "Instituto de Pesquisas Geológicas que fica";
					mes "no segundo andar da Loja de Armas.";
					if (.@compass_check == 1) viewpoint 1,150,175,7,0x00FF00;
					break;
				case 8:
					mes "[Guia de Veins]";
					mes "Se você quiser se livrar de todas as indicações marcadas no seu Mini-mapa,";
					mes "apenas me procure, e escolha a opção ''Remover Marcações do Mini-mapa'' no menu.";
					set .@loop2,0;
					break;
				}
			}
			break;

		case 2:
			viewpoint 2,196,258,1,0xFF0000;
			viewpoint 2,128,266,2,0xFF00FF;
			viewpoint 2,150,175,3,0x99FFFF;
			viewpoint 2,230,161,4,0x0000FF;
			viewpoint 2,273,285,5,0x00FF00;
			viewpoint 2,150,217,6,0x00FF00;
			viewpoint 2,150,175,7,0x00FF00;
			mes "[Guia de Veins]";
			mes "Ok, as marcações já sumiram. Se você quiser informações sobre outros lugares, é só falar comigo.";
			mes "Aproveite sua estadia em Veins.";
			break;
		case 3:
			mes "[Guia de Veins]";
			mes "Quando você estiver usando o menu ''Guias da Cidade'',";
			mes "certifique-se de que as indicações de locais foram marcadas no seu Mini-mapa no canto superior direito da tela.";
			mes "Se você não consegue visualizar o seu Mini-mapa, use as teclas de atalho ''Ctrl + Tab'', ou pressione o botão ''Mapa-múndi'' no menu da sua janelas de Informações Básicas, ok?";
			mes "E você também pode ampliar o seu Mini-mapa usando o botão ''-'' caso não consiga ver o mapa inteiro da vila.";
			break;
		case 4:
			mes "[Guia de Veins]";
			mes "Aproveite sua estadia em Veins.";
			close2;
			set .@loop1,0;
			break;
		}
	}
	close;
}

veins,189,101,5	duplicate(ve_guide)	Guia de Veins#2	934
