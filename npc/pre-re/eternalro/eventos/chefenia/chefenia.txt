-	script	Chefenia_Core#PREMIUM	-1,{

OnInit:
	// Configuração
	$CHEFENIA_ON = 0;
	$CHEFENIA_ITEM = 7227;

	// Pontos por MVP
	$CHEFENIA_POINTS = 5;

	// Horário (CONFIGURÁVEL)
	$CHEFENIA_START = 13; // abre
	$CHEFENIA_END   = 22; // fecha

	// Limpa ranking
	deletearray $@chef_name$;
	deletearray $@chef_points;

	initnpctimer;
	end;

OnTimer60000:

	.@hour = gettime(3);

	// ABRIR TODO DIA
	if (.@hour >= $CHEFENIA_START && .@hour < $CHEFENIA_END) {

		if (!$CHEFENIA_ON) {
			$CHEFENIA_ON = 1;
			announce "[Chefenia] Evento ATIVO!", bc_all;
		}

	} else {

		if ($CHEFENIA_ON) {
			$CHEFENIA_ON = 0;
			announce "[Chefenia] Evento ENCERRADO!", bc_all;
		}
	}

	initnpctimer;
	end;

}

-	script	Chefenia_System#PREMIUM	-1,{

OnNPCKillEvent:

	if (!$CHEFENIA_ON) end;

	if (!(strcharinfo(3) == "bossnia_01" || strcharinfo(3) == "bossnia_02" || strcharinfo(3) == "bossnia_03" || strcharinfo(3) == "bossnia_04"))
		end;

	if (getmonsterinfo(killedrid, MOB_MVPEXP) <= 0)
		end;

	// Pontos
	#chef_points += $CHEFENIA_POINTS;
	#chef_kills++;

	// HUD
	dispbottom "[Chefenia] MVPs: "+#chef_kills+" | Pontos: "+#chef_points;

	// Drop random
	.@r = rand(100);

	if (.@r < 5) {
		getitem 12299,1;
		announce "[Chefenia] "+strcharinfo(0)+" conseguiu item LENDÁRIO!", bc_all;
	}
	else if (.@r < 25) {
		getitem 607,2;
	}
	else {
		getitem 909,5;
	}

	// Atualiza ranking
	callsub S_UpdateRanking;

	end;

// Ranking global
S_UpdateRanking:

	for (.@i = 0; .@i < 10; .@i++) {

		if ($@chef_points[.@i] < #chef_points) {

			// shift
			for (.@j = 9; .@j > .@i; .@j--) {
				$@chef_points[.@j] = $@chef_points[.@j-1];
				$@chef_name$[.@j] = $@chef_name$[.@j-1];
			}

			$@chef_points[.@i] = #chef_points;
			$@chef_name$[.@i] = strcharinfo(0);
			break;
		}
	}

	return;

}
