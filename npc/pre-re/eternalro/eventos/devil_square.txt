/*= Type 2 [Hard-Mode]
	Description: 
	In this version when you die and get warped out, if you hurry up you'll be able to come back into Devil Square again. 
	If you don't hurry up and repay to join back into the event, someone else could have taken your spot, so be quick about getting back in!
	Since players can die, and repay to come back inside the event, there's no need to cancel out the event if no one joins. (Which is in Type 1, once all players die that attempted the Devil's Square the event is ended.)
	NOTE: You WILL have to REPAY the amount of Zeny to re-enter.
	
/*= Difference in Type 1:
	In this version of Devil's Square, once the amount of players have joined, that's it. 
	You won't be able to continue going back inside of Devil Square once you die, after entering and Devil Square starts, it will be closed. 
	All other players won't be able to join, and it's up to all the players to either complete Devil Square or die and not be able to come back in.
	If no players enter in the timeframe, it will cancel out the event.
	Also counts up the Max amount of Users when they enter Devil's Square. Once the Max Users have been established, no one else can join, and then it's just up to that team of people!
	Once all the Players have died and warped out of the event, it will announce that all the users who entered have failed Victory in Devil Square, and the event will be reset.

	
	FINAL NOTE: You can't use BOTH  Type 1 and Type 2, as they occur on the same map (This could be changed in the future for Easy/Hardcore modes as long as I make or copy the same map with different variable names.
	
/*= External Functions:
	None

/*= New Additions to Revision 6
	[Revision 6]
	Fixed a major bug where if the event was active or not that if you died you would automatically be sent to your SavePoint, other minor things addressed too. (Hard-Mode version)
	[Revision 5]
	Created commands:
	@disableds - Allows you to disable Devil Square whenever you want (This feature is also inside of the NPC!)
	@repeatds - Allows you to repeat the Devil Square on the go. (This feature is also inside of the NPC!)
	@editds -  Allows you to edit, or as I put it "Over Ride" the settings in the OnInit: inside of [DS] Organizer::DSORG.
		Features that are editable via `@editds`:
		
	Changes happen in the menu whether the event is active or not. If it's active it will allow you to disable or go to the Player Menu, if it's unactive it'll ask if you want to Start Devil's Square.
	Basically redesigned a lot of the aspects of the script, beforehand it didn't even work, and had errors upon errors.
	Made the script to be flexible and totally configurable!
	Future additions/ideas: Add the ability to also edit the mob data for each Round as well as the MvP list.
	Also before hand if it was <= 5 users it would only spawn 8 Treasure Chests, otherwise it would spawn 16. Now you can choose as much as you want!
	When a GM is editing an option inside of Devil's Square, another GM cannot edit an option until that GM is done editing an option via `@editds`.
	
	NOTE: Not every Treasure Chest is the same either!
=*/

/*= DS Organizer =*/
que_temsky,39,146,4	script	[DS] Organizer::DSORG	4_GEFFEN_11,{
	$npc$ = "[ ^FF0000Devil's Square^000000 ^E9AB17Organizer^000000 ]";
	
	if(getgmlevel() >= $@MinGM) {
		callsub GM_Menu;
		end;
	}

OnPlayerMenu:
	mes $npc$;
	mes "Bem-vindo ao nosso evento especial!!";
	mes "Somente ^0000FF"+$@ZenReq+"^000000 Zeny para voce desafiar este maravilhoso mundo das tribulações!";
	next;
	while(1)
	{
		mes $npc$;
		mes "Então, você gostaria de participar e tentar a sorte?";
		mes "Observação: você deve ter pelo menos nível ^FF0000"+$@MinBaseLevel+"^000000 para entrar.";
		switch(select("Claro!:Não obrigado, Eu valorizo minha vida:Hum, no que estou me metendo?"))
		{
			case 1:
				if( !$@DS_Open )
				{
					mes "Sinto muito mas o Devil Square nao esta aberto";
					mes "Volte em outro momento.";
					close;
				}
				if( BaseLevel < $@MinBaseLevel || Zeny < $@ZenReq )
				{
					mes "Sinto muito mas você precisa ter nivel no minimo de ^FF0000"+$@MinBaseLevel+"^000000 ou você nao tem ^FF0000"+$@ZenReq+"^000000 Zeny.";
					close;
				}
				if(getmapusers("ordeal_1-1") >= $@MaxUsers )
				{
					mes "Sinto muito, mas estamos com capacidade máxima. Por favor, tente novamente na próxima vez.";
					close;
				}
				Zeny-=$@ZenReq;
				announce strcharinfo(0) + " Decidiu enfrentar o Devil Square em busca de fama e gloria! Desejo " + ((Sex)?"him":"her") + " Sorte!",8;
				percentheal 100,100;
				warp "ordeal_1-1",149,149;
				end;
			
			case 2:
				mes "Lamento ouvir isso. Por favor, volte se mudar de ideia.";
				close;
			
			case 3:
				next;
				mes $npc$;
				mes "Devil's Square é um evento do modo horda que consiste em 10 rodadas.";
				mes "As rodadas começam fáceis, mas ficam cada vez mais difíceis.";
				mes "A rodada final consiste em um único mob MVP.";
				next;
				mes $npc$;
				mes "Uma vez aberta a porta, você tem "+$@MaxMin+" minutos para entrar.";
				mes "Se você não entrar a tempo, você perde essa rodada.";
				mes "Se todos os participantes morrerem, a rodada será cancelada.";
				next;
				break;
		}
	}
	
GM_Menu:
	mes $npc$;
	mes "^0000FF--- STATUS ATUAL DO EVENTO ---^000000";
	mes "Status: " + (($@DS_Active)?"^00FF00ATIVO^000000":"^FF0000INATIVO^000000");
	mes "Requisito mínimo de GM: ^0000FF"+$@MinGM+"^000000";
	mes "Máximo de Usuários: ^0000FF"+$@MaxUsers+"^000000";
	mes "Base Level Mínimo: ^0000FF"+$@MinBaseLevel+"^000000";
	mes "Max minutos para entrar: ^0000FF"+$@MaxMin+"^000000";
	mes "Qtd máxima de baús: ^0000FF"+$@MaxChests2+"^000000";
	mes "Zeny Requerido: ^0000FF"+$@ZenReq+"^000000";
	mes "Ativar Drops: ^0000FF"+$@EnableDrops+"^000000";
	next;
	
	if ( $@DS_Active ) {
		switch(select("? Desativar Devil's Square", "?? Configurar Opções", "?? Entrar como Player")) {
			case 1:
				if ( !$ChestsOverRide ) { $@MaxChests = $@MaxChests2; } else { $@MaxChests = $MaxChests2; }
				if ( !$@EnableDrops ) { removemapflag "ordeal_1-1",mf_nomobloot; }
				$@DS_Open = 0; $@DS_Active = 0;
				killmonsterall "ordeal_1-1";
				announce "[GM] "+strcharinfo(0)+" Desativou o Devil's square.",bc_map;
				mapwarp "ordeal_1-1","prontera",150,150;
				close;
			case 2:
				goto OnLocalEditDS;
			case 3:
				goto OnPlayerMenu;
		}
	} else {
		switch(select("?? Iniciar Devil's Square", "?? Configurar Opções", "?? Entrar como Player")) {
			case 1:
				removemapflag "ordeal_1-1",mf_noskill;
				if ( !$@EnableDrops ) { setmapflag "ordeal_1-1",mf_nomobloot; }
				if ( $@EnableDrops ) { removemapflag "ordeal_1-1",mf_nomobloot; }
				donpcevent ("DS_EVENT#DSMAIN")+"::OnStartDS";
				end;
			case 2: 
				goto OnLocalEditDS;
			case 3:
				goto OnPlayerMenu;
		}
	}
	return;

OnLocalEditDS:
	mes "[ Editor In-Game ]";
	mes "Escolha o que deseja alterar:";
	next;
	switch(select("?? Min GM Req ("+$@MinGM+")","?? Max Users ("+$@MaxUsers+")","?? Min Level ("+$@MinBaseLevel+")","?? Tempo de Entrada ("+$@MaxMin+" min)","?? Baús de Tesouro ("+$@MaxChests2+")","?? Zeny Req ("+$@ZenReq+")","?? Ativar Drops ("+$@EnableDrops+")","?? Voltar ao Menu Principal"))
	{
		case 1:
			mes "Digite o novo Nivel de GM minimo (20-99):";
			input $MinGM;
			if($MinGM < 20 || $MinGM > 99) { dispbottom "Valor Invalido!"; break; }
			$@MinGM = $MinGM; $MinGMOverRide = 1;
			break;
		case 2:
			mes "Digite o Maximo de Usuarios permitidos:";
			input $MaxUsers;
			if($MaxUsers <= 0) { dispbottom "Valor deve ser maior que 0!"; break; }
			$@MaxUsers = $MaxUsers; $MaxUserOverRide = 1;
			break;
		case 3:
			mes "Digite o Level minimo para entrar:";
			input $MinBaseLevel;
			if($MinBaseLevel <= 0) { dispbottom "Valor Invalido!"; break; }
			$@MinBaseLevel = $MinBaseLevel; $MinBaseOverRide = 1;
			break;
		case 4:
			mes "Digite o tempo de entrada (em minutos):";
			input $MaxMin;
			if($MaxMin <= 0) { dispbottom "Valor Invalido!"; break; }
			$@MaxMin = $MaxMin; $MaxMinOverRide = 1;
			break;
		case 5:
			mes "Digite a quantidade de baús:";
			input $MaxChests2;
			if($MaxChests2 <= 0) { dispbottom "Valor Invalido!"; break; }
			$ChestsOverRide = 1; $@MaxChests = $MaxChests2; $@MaxChests2 = $MaxChests2;
			break;
		case 6:
			mes "Digite o valor de Zeny cobrado:";
			input $ZenReq;
			if($ZenReq < 0) { dispbottom "Valor Invalido!"; break; }
			$@ZenReq = $ZenReq; $ZenReqOverRide = 1;
			break;
		case 7:
			mes "Habilitar drop global de monstros comuns?";
			mes "1 - Sim | 0 - Nao (Apenas MVP/Baús dropam)";
			input $EnableDrops;
			if($EnableDrops != 0 && $EnableDrops != 1) { dispbottom "Apenas 0 ou 1!"; break; }
			$@EnableDrops = $EnableDrops; $EnableDropsOverRide = 1;
			break;
		case 8:
			goto GM_Menu;
	}
	next;
	goto OnLocalEditDS;

OnInit:
	$@DSMapName$ = "prontera"; 
	if ( !$MinGMOverRide ) { $@MinGM = 99; } else { $@MinGM = $MinGM; }
	if ( !$ZenReqOverRide ) { $@ZenReq = 10000; } else { $@ZenReq = $ZenReq; }
	if ( !$MinBaseOverRide ) { $@MinBaseLevel = 50; } else { $@MinBaseLevel = $MinBaseLevel; }
	if ( !$MaxUserOverRide ) { $@MaxUsers = 10; } else { $@MaxUsers = $MaxUsers; }
	if ( !$MaxMinOverRide ) { $@MaxMin = 5; } else { $@MaxMin = $MaxMin; }
	if ( !$ChestsOverRide ) {
		$@MaxChests = 2;
		$@MaxChests2 = 2;
	} else {
		$@MaxChests = $MaxChests2;
		$@MaxChests2 = $MaxChests2;
	}
	if ( !$EnableDropsOverRide ) { $@EnableDrops = 0; } else { $@EnableDrops = $EnableDrops; }
	end;
}

/*= Event Code =*/
ordeal_1-1,0,0,0	script	DS_EVENT#DSMAIN	-1,{

OnInit:
	bindatcmd "repeatds",strnpcinfo(0)+"::OnAtRepeatDS";
	bindatcmd "disableds",strnpcinfo(0)+"::OnAtDisableDS";
	$@DS_Open = 0;
	setarray $@DS1,1002,1113,1005,1010,1090,1089,1089,1089,1015,1076,1015,1094,1025;
	setarray $@DS2,1068,1052,1019,1077,1089,1088,1018,1028,1176,1128,1056,1169;
	setarray $@DS3,1073,1057,1057,1686,1138,1212,1629,1323,1074,1016,1144,1317;
	setarray $@DS4,1248,1099,1129,1880,1881,1151,1264,1108,1409,1069,1045,1271;
	setarray $@DS5,1101,1061,1412,1291,1109,1699,1299,1065,1165,1180,1127;
	setarray $@DS6,1302,1865,1297,1752,1753,1870,1283,1773,1773,1989,1989;
	setarray $@DS7,1678,1679,1705,1704,1707,1706,1867,1867,1867,1867;
	setarray $@DS8,1678,1678,1679,1117,1117,1291,1200,1291,1291,1838;
	setarray $@DS9,1993,1992,1986,2016,2018,2017,1268,2017,1263,1262;
	setarray $@DSMVP,1511,1039,1418,1768,1373,2022,1087,1190,1157,1623,1312,1685,1159,1147,1059,1150,1492,1251,1252,1272;
	setarray $@DSC,132,137,167,159, 54,54,110,110, 26,138,95,182, 109,189,54,245, 138,204,162,273, 190,190,246,246, 208,137,273,181, 189,110,245,52, 137,90,161,26;
	end;

OnStartDS:
	$@DS_Active = 1;
	$@DS_Open = 1;
	if ( !$ChestsOverRide ) { $@MaxChests = $@MaxChests2; } else { $@MaxChests = $MaxChests2; }
	mapwarp "ordeal_1-1","prontera",150,150;
	killmonsterall "ordeal_1-1";
	announce "[Devil's Square] Os portais para o Devil's Square Estao ABERTOS! Voce tem "+$@MaxMin+" minutos para entrar antes de comecar!",0;
	for(.@i = $@MaxMin; .@i  > 0; .@i--)
	{
		announce "Devil Square comecara em " + .@i + " minutos!", 0;
		sleep 60000;
	}
	$@DS_Round = 1;
	donpcevent strnpcinfo(0)+"::OnBeginRound";
	announce "[Devil's Square] Devil Square COMECOU!",0;
	end;

OnBeginRound:
	if($@DS_Round == 10)
	{
		mapannounce "ordeal_1-1","[Devil's Square] Va para o centro do mapa para a rodada do chefe.", 0;
		while(getareausers("ordeal_1-1", 142, 142, 157, 157) < getmapusers("ordeal_1-1"))
			sleep 25;
		.@MVP = $@DSMVP[rand(getarraysize($@DSMVP))];
		areamonster "ordeal_1-1", 132, 137, 167, 159, "[DS]" + getmonsterinfo(.@MVP, MOB_NAME), .@MVP, 1, "DS_EVENT#DSMAIN::OnDSMVPDead";
		end;
	}
	while(getarraysize($@RoundMobs) < 5) {
		.@mob = getd("$@DS" + $@DS_Round + "[" + rand(getarraysize(getd("$@DS" + $@DS_Round))) + "]");
		.@dup = 0;
		for(.@j = 0; .@j < getarraysize($@RoundMobs); .@j++) {
			if(.@mob == $@RoundMobs[.@j]) .@dup = 1;
		}
		if(!.@dup) $@RoundMobs[getarraysize($@RoundMobs)] = .@mob;
	}
	for(.@i = 0; .@i < getarraysize($@RoundMobs); .@i++)
		areamonster "ordeal_1-1",$@DSC[0 + (4 * ($@DS_Round - 1))], $@DSC[1 + (4 * ($@DS_Round - 1))], $@DSC[2 + (4 * ($@DS_Round - 1))], $@DSC[3 + (4 * ($@DS_Round - 1))], getmonsterinfo($@RoundMobs[.@i], MOB_NAME), $@RoundMobs[.@i], rand(3,5) * (((11 - $@DS_Round) / 2) + (12 - $@DS_Round)), strnpcinfo(0)+"::OnDSDead";
	mapannounce "ordeal_1-1","Rodada " + $@DS_Round + " comecou!", 0;
	deletearray $@RoundMobs,getarraysize($@RoundMobs);
	end;

OnDSDead:
	switch(mobcount("ordeal_1-1","DS_EVENT#DSMAIN::OnDSDead"))
	{
		case 200: case 150: case 100: case 50: case 25: case 10: case 5:
			mapannounce "ordeal_1-1","Monstros restantes: " + mobcount("ordeal_1-1","DS_EVENT#DSMAIN::OnDSDead"),0;
			break;
		case 0:
			$@DS_Round++;
			mapannounce "ordeal_1-1","Todos os monstros da rodada " + ($@DS_Round - 1) + " foram mortos!",0;
			donpcevent strnpcinfo(0)+"::OnBeginRound";
		default:
			if(mobcount("ordeal_1-1","DS_EVENT#DSMAIN::OnDSDead") < 5) {
				mapannounce "ordeal_1-1","Ate o ultimo "+ mobcount("ordeal_1-1","DS_EVENT#DSMAIN::OnDSDead"),0;
			}
	}
	end;

OnDSMVPDead:
	mapannounce "ordeal_1-1", "Parabens! Agora voce pode abrir suas caixas de tesouro!", 0;
	setmapflag "ordeal_1-1",mf_noskill;
	for(.@i = 0; .@i  < $@MaxChests; .@i++) {
		monster "ordeal_1-1",0,0, "[DS] Treasure Box",rand(1324,1356),1,"DS_EVENT#DSMAIN::OnBoxOpened";
	}
	end;

OnBoxOpened:
	$@MaxChests-=1;
	if (!$@MaxChests) {
		announce "[Devil's Square] " + strcharinfo(0) + " abriu a ultima Caixa do Tesouro. Teleportando...", 0;
		if ( !$@EnableDrops ) { removemapflag "ordeal_1-1",mf_nomobloot; }
		sleep 3000;
		$@DS_Active = 0; $@DS_Open = 0;
		mapwarp "ordeal_1-1","prontera",150,155;
		end;
	}
	announce "[Devil's Square] " + strcharinfo(0) + " Abriu uma Caixa do Tesouro! ("+$@MaxChests+") restantes!", 0;
	end;

OnPCDieEvent:
	if(strcharinfo(3) == "ordeal_1-1") {
		announce strcharinfo(0) +" Acabou de morrer no Devil's Square! Vaga disponivel!",bc_all;
		warp "SavePoint",0,0;
	}
	end;

OnAtRepeatDS:
	if (getgmlevel() < 99) end;
	if ( $@DS_Active ) { dispbottom "Atualmente Devil Square esta ativado."; end; }
	donpcevent ("DS_EVENT#DSMAIN")+"::OnStartDS";
	if ( !$@EnableDrops ) { setmapflag "ordeal_1-1",mf_nomobloot; }
	end;

OnAtDisableDS:
	if (getgmlevel() < 99) end;
	if ( !$@DS_Active ) { dispbottom "Atualmente Devil Square nao esta ativado."; end; }
	if ( !$ChestsOverRide ) { $@MaxChests = $@MaxChests2; } else { $@MaxChests = $MaxChests2; }
	if ( !$@EnableDrops ) { removemapflag "ordeal_1-1",mf_nomobloot; }
	$@DS_Open = 0; $@DS_Active = 0;
	killmonsterall "ordeal_1-1";
	announce "[GM] "+strcharinfo(0)+" Desativou o Devil's square.",bc_map;
	mapwarp "ordeal_1-1","prontera",150,150;
	end;
}
