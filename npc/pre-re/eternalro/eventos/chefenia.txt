function	script	F_CheffeniaRateDiscount	{
    .@discount = 0;
    switch (CUSTOM_RATE) {
        case 25: .@discount = 15; break;
        case 12: .@discount = 30; break;
        case 1:  .@discount = 60; break;
        case 6:  .@discount = 70; break;
    }
    if (vip_status(VIP_STATUS_ACTIVE) && .@discount < 10)
        .@discount = 10;
    return .@discount;
}

    mes "[Cheffenia]";
    mes "Modo escolhido: " + .@difficulty$ + ".";
    mes "Acesso: " + .@zeny_text$ + " zenys ou " + .@vote_cost + " ponto(s) de voto.";

    if (.@payment == 3 || .@payment == 255) close;

    if (.@payment == 1) {
        if (Zeny < .@zeny_cost) {
            mes "[Cheffenia]";
            mes "Voce nao tem zenys suficientes.";
            close;
        }
        if (select("Confirmar pagamento de " + .@zeny_text$ + " zenys:Cancelar") != 1) close;
        Zeny -= .@zeny_cost;

        if (#VOTEPOINTS < .@vote_cost) {
            mes "[Cheffenia]";
            mes "Voce nao tem pontos de voto suficientes.";
            close;
        }
        if (select("Confirmar pagamento de " + .@vote_cost + " ponto(s) de voto:Cancelar") != 1) close;
        #VOTEPOINTS -= .@vote_cost;
    }

    close2;
    warp .@map$, rand(202,204), rand(202,204);
    end;
}
