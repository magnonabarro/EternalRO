eabyss_03h,0,0,0	script	#TurnIn_Drop	-1,{

OnTurnInMobDead:
if (BaseLevel <= 151) {
	 goto OnModDeathTurnInDrop; 
	 } 
	 else { 
		 dispbottom "[ Seu Servidor ] : Você já atingiu o nível máximo recomendado para este mapa.";
		 sleep2 500;
		 warp "SavePoint",0,0;
		 end; }

OnModDeathTurnInDrop:
if(getcharid(1)) {
    getpartymember getcharid(1),1;
    getpartymember getcharid(1),2;
    
    for ( .@p = 0; .@p < $@partymembercount; .@p++)
        if(attachrid($@partymemberaid[.@p],$@partymembercid[.@p]) == 1) {
            if(strcharinfo(3) != strnpcinfo(4)) continue;
            if(rand(1,10000) >= (10000 - .chance)) 
                getitem .drop_id[rand(getarraysize(.drop_id))],1;
        }

} else {
	if(rand(1,10000) >= (10000 - .chance)) 
		getitem .drop_id[rand(getarraysize(.drop_id))],1;
}
end;

OnInit:
setarray .drop_id[0],60003; // Moeda Turn In

// ==== Drop Settings ====
// Multiply the Drop Chance below with the Server Drop Chance
// 1 = Yes
// 0 = No
.battleconf_drop = 0;

// General Drop Chance
.drop = 400; // 50 = 0.5%

// Calculation
.chance = .drop * ( (.battleconf_drop)?( (getbattleflag("item_rate_common")%100 > 0)? (getbattleflag("item_rate_common")/100):1):1);
end;
}
