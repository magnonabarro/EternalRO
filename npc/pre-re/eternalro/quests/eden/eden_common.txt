// Entrega moedas ao concluir uma missão do Grupo Éden.
// CONFIGURAÇÃO: substitua 0 pelo ID do item usado como moeda no seu servidor.
// A quantidade é informada pelo script do quadro conforme a dificuldade da missão.
function	script	F_EdenMissionCoinReward	{
	.@coin_id = 25223; // <<< COLOQUE AQUI O ID DA MOEDA
	.@amount = getarg(0,1);
	if (.@coin_id > 0 && .@amount > 0) {
		getitem .@coin_id,.@amount;
		dispbottom "[Grupo Éden] Você recebeu "+.@amount+" moeda(s) pela conclusão da missão.";
	}
	else {
		dispbottom "[Grupo Éden] Configure o ID da moeda em F_EdenMissionCoinReward no arquivo eden_common.txt.";
	}
	return;
}
