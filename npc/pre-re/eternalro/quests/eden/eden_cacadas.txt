function	script	F_EdenHuntWarp	{



	.@i=getarg(0);



	.@tier=.@i/5;



	.@qid=getvariableofnpc(.Quest[.@i],"EdenCacadas");



	.@state=checkquest(.@qid);



	if (BaseLevel<getvariableofnpc(.Min[.@tier],"EdenCacadas") || (.@state!=0 && .@state!=1)) {



		mes "Aceite uma caçada liberada para seu nível antes de teleportar.";



		close;



	}



	.@map$=getvariableofnpc(.Map$[.@i],"EdenCacadas");



	if (.@map$=="" || getmapflag(.@map$,mf_nowarpto) || getmapflag(strcharinfo(3),mf_nowarp)) {



		mes "O teleporte para este destino está indisponível neste momento.";



		close;



	}



	mes "[Teleporte de Caça]";



	mes "Alvo: "+getvariableofnpc(.Name$[.@i],"EdenCacadas")+".";



	mes "Mapa: ^0055FF"+.@map$+"^000000.";



	mes "População configurada: "+getvariableofnpc(.MapCount[.@i],"EdenCacadas")+" monstros deste alvo.";



	mes "Este é o destino indicado para esta caçada. O teleporte é gratuito.";



	next;



	if (select("Teleportar para o mapa de caça:Continuar no Éden")==2) close;



	close2;



	warp .@map$,0,0;



	end;



}

// Configuracao de recompensas e catalogo: OnInit de EdenCacadas.



function	script	F_EdenCacadas	{

	.@tier = getarg(0);

	.@min = getvariableofnpc(.Min[.@tier],"EdenCacadas");

	mes "[Caçadas do Éden]";

	mes "Categoria: ^0055FF"+getvariableofnpc(.TierName$[.@tier],"EdenCacadas")+"^000000.";

	mes "Nível mínimo: ^0055FF"+.@min+"^000000. Sem limite máximo.";

	if (BaseLevel < .@min) {

		mes "Você ainda não tem o nível necessário para estas caçadas.";

		close;

	}

	if (!callfunc("F_HasEdenGroupMark")) {

		mes "Entre para o Grupo Éden com a secretária Lime Evenor antes de aceitar uma caçada.";

		close;

	}

	.@total = getvariableofnpc(.Total,"EdenCacadas");

	.@active = -1;

	for (.@i=0; .@i<.@total; .@i++) {

		.@qid = getvariableofnpc(.Quest[.@i],"EdenCacadas");

		if (checkquest(.@qid) == 0 || checkquest(.@qid) == 1) {

			.@active = .@i;

			break;

		}

	}

	if (.@active >= 0) {

		.@i = .@active;

		.@qid = getvariableofnpc(.Quest[.@i],"EdenCacadas");

		.@mob = getvariableofnpc(.Mob[.@i],"EdenCacadas");

		.@count = getvariableofnpc(.Count[.@i],"EdenCacadas");

		.@name$ = getvariableofnpc(.Name$[.@i],"EdenCacadas");

		.@questtier = .@i / 5;

		mes "Caçada atual: ^0055FF"+.@name$+"^000000.";

		mes "Objetivo: derrotar "+.@count+" monstros.";

		mes "Confira a contagem na janela de missões (Alt+U).";

		next;

		if (checkquest(.@qid,HUNTING) == 2) {

			mes "[Caçadas do Éden]";

			mes "Caçada concluída!";

			.@coins = getvariableofnpc(.Coins[.@questtier],"EdenCacadas");

			.@coin = getvariableofnpc(.Coin,"EdenCacadas");

			.@bexp = getvariableofnpc(.BaseExp[.@questtier],"EdenCacadas");

			.@jexp = getvariableofnpc(.JobExp[.@questtier],"EdenCacadas");

			mes "Recompensa: "+.@coins+" moedas do Éden, "+.@bexp+" EXP base e "+.@jexp+" EXP de classe.";

			if (select("Receber recompensa:Voltar depois") == 2) close;

			if (!checkweight(.@coin,.@coins) || countitem(.@coin)+.@coins > 30000) {

				mes "Libere espaço e peso no inventário. Sua recompensa continua disponível.";

				close;

			}

			// Complete antes da entrega para impedir resgates repetidos.

			completequest .@qid;

			.@cool = getvariableofnpc(.Cooldown[.@i],"EdenCacadas");

			if (checkquest(.@cool) >= 0) erasequest .@cool;

			setquest .@cool;

			getitem .@coin,.@coins;

			getexp .@bexp,.@jexp;

			mes "Use suas moedas no NPC Troca do Eden.";

			mes "Esta caçada poderá ser repetida após 3 horas. As outras continuam disponíveis.";

			close;

		}

		mes "[Caçadas do Éden]";

		switch(select("Continuar a caçada:Teleportar para a caçada:Cancelar a caçada")) {

		case 1:

			mes "Derrote "+.@count+" "+.@name$+" e volte a qualquer quadro de caçadas do Éden.";

			mes "Você pode usar @whereis "+.@mob+" para procurar os mapas do monstro.";

			close;

		case 2:

			callfunc "F_EdenHuntWarp",.@i;

			close;

		case 3:

			mes "O progresso desta caçada será perdido. Deseja cancelar?";

			next;

			if (select("Manter a caçada:Confirmar cancelamento") == 2) {

				erasequest .@qid;

				mes "Caçada cancelada. Você pode escolher outra.";

			}

			close;

		}

	}

	mes "Escolha uma caçada. Você pode fazer as categorias de nível menor que o seu.";

	mes "Todas as missões deste quadro têm a dificuldade desta categoria.";

	mes "Recompensa por caçada: ^0055FF"+getvariableofnpc(.Coins[.@tier],"EdenCacadas")+" moedas do Éden^000000 + EXP.";

	next;

	.@menu$ = "";

	for (.@n=0; .@n<5; .@n++) {

		.@i = .@tier*5+.@n;

		.@cool = getvariableofnpc(.Cooldown[.@i],"EdenCacadas");

		.@wait = checkquest(.@cool,PLAYTIME);

		.@menu$ += getvariableofnpc(.Name$[.@i],"EdenCacadas")+" ("+getvariableofnpc(.Count[.@i],"EdenCacadas")+")";

		if (.@wait == 0 || .@wait == 1) .@menu$ += " [Aguardando]";

		.@menu$ += ":";

	}

	.@selected = select(.@menu$+"Sair")-1;

	if (.@selected == 5) close;

	.@i = .@tier*5+.@selected;

	.@qid = getvariableofnpc(.Quest[.@i],"EdenCacadas");

	.@cool = getvariableofnpc(.Cooldown[.@i],"EdenCacadas");

	.@wait = checkquest(.@cool,PLAYTIME);

	mes "[Caçadas do Éden]";

	if (.@wait == 0 || .@wait == 1) {

		mes "Esta caçada está em espera. Aguarde 3 horas a partir da última entrega ou escolha outro monstro.";

		close;

	}

	mes "Alvo: ^0055FF"+getvariableofnpc(.Name$[.@i],"EdenCacadas")+"^000000.";

	mes "Quantidade: "+getvariableofnpc(.Count[.@i],"EdenCacadas")+".";

	mes "Moedas do Éden: "+getvariableofnpc(.Coins[.@tier],"EdenCacadas")+".";

	mes "EXP base: "+getvariableofnpc(.BaseExp[.@tier],"EdenCacadas")+". EXP de classe: "+getvariableofnpc(.JobExp[.@tier],"EdenCacadas")+".";

	mes "Somente monstros derrotados após aceitar a missão serão contados.";

	next;

	if (select("Aceitar caçada:Cancelar") == 2) close;

	if (BaseLevel < .@min) close;

	// Revalida antes de criar a missao; o progresso fica exclusivamente no quest_db.

	for (.@n=0; .@n<.@total; .@n++) {

		.@other = getvariableofnpc(.Quest[.@n],"EdenCacadas");

		if (checkquest(.@other) == 0 || checkquest(.@other) == 1) {

			mes "Você já possui uma caçada em andamento.";

			close;

		}

	}

	if (checkquest(.@cool) >= 0) erasequest .@cool;

	if (checkquest(.@qid) >= 0) erasequest .@qid;

	setquest .@qid;

	mes "Caçada iniciada! Acompanhe a contagem em Alt+U.";

	mes "Ao terminar, volte a um quadro do Éden para receber suas moedas.";

	callfunc "F_EdenHuntWarp",.@i;

	close;

}
