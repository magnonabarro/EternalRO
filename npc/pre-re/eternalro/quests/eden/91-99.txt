L_Quest:
	set .@quest1, getarg(0);
	set .@quest2, .@quest1+31;
	.@cutin$ = getarg(3);
	set .@playtime, checkquest(.@quest2,PLAYTIME);
	if (.@playtime == 0 || .@playtime == 1) {
		mes "[Quadro de Missões 91-99]";
		mes "Você precisa esperar 3 horas antes de aceitar esta missão novamente.";
		close;
	}
	set .@hunting, checkquest(.@quest1,HUNTING);
	if (.@hunting == 0 || .@hunting == 1) {
		mes "[Quadro de Missões 91-99]";
		mes "Você já derrotou os 30 "+getarg(1)+getarg(5,"")+"?";
		close;
	} else if (.@hunting == 2) {
		mes "[Quadro de Missões 91-99]";
		mes "Você concluiu a missão de caça.";
		mes "Aqui está sua recompensa pela conclusão da missão.";
		erasequest .@quest1;
		if (.@playtime > -1) erasequest .@quest2;
		setquest .@quest2;

		// Cálculo da experiência de classe.
		if (JobLevel > 50 && JobLevel < 56) set .@jexp,67000;
		else if (JobLevel > 55 && JobLevel < 61) set .@jexp,94000;
		else if (JobLevel > 60 && JobLevel < 66) set .@jexp,141000;
		else if (JobLevel > 65 && JobLevel < 69) set .@jexp,212000;
		else if (JobLevel == 69) set .@jexp,311000;

		getexp getarg(2),.@jexp;
		callfunc("F_EdenMissionCoinReward",7);
		close;
	}/* else if (.@hunting == 3) {
//		recall_completequest .@quest1
		erasequest .@quest1;
		mes "[Quadro de Missões 91-99]";
		mes "Você aceitou novamente a missão de caça: "+getarg(1)+".";
		close;
	}*/ else {
		mes "[Quadro de Missões 91-99]";
		mes getarg(4);
		next;
		if (.@cutin$ != "") cutin .@cutin$,3;
		mes "[Quadro de Missões 91-99]";
		mes "Cace 30 monstros "+getarg(1)+" nessa região e registre a conclusão neste quadro.";
		next;
		if (.@cutin$ != "") cutin .@cutin$,255;
		if(select("Vou caçá-los.:Não, obrigado.") == 1) {

			mes "Quando concluir a missão, registre-a neste quadro e receba sua recompensa.";
			if (.@playtime > -1) erasequest .@quest2;
			setquest .@quest1;
		}
		close;
	}
