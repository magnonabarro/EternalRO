//===== Autor: ===============================================

//===== Versão atual: ========================================
//= 1.4
//===== Compatibilidade: =====================================
//= Projeto rAthena
//===== Descrição: ===========================================
//= Missões repetíveis para personagens do nível base 41 a 55.
//===== Histórico de alterações: =============================
//= 1.0 Primeira versão.
//= 1.1 Pequenas otimizações. [Masao]
//= 1.2 Otimizações. [Euphy]
//= 1.3 Adicionado suporte a Para_Team_Mark_ no Grupo Éden. [Ragno]
//= 1.4 Adicionado uso da função F_HasEdenGroupMark. [Ragno]
//============================================================

moc_para01,40,38,3	script	Mission [41 - 55]	857,{
	if (!callfunc("F_HasEdenGroupMark")) {
		mes "Você não é membro do Grupo Éden e não tem autorização para acessar este quadro de missões.";
		close;
	}

	if (checkweight(1201,1) == 0 || (MaxWeight - Weight) < 1000) {
		mes "^008800Espere um momento!!";
		mes "Você não pode receber mais itens porque está carregando peso demais. Libere espaço no inventário e tente novamente.^000000";
		close;
	}

	if (checkquest(12088) == -1) {
		if (BaseLevel < 41 || BaseLevel > 55) {
			mes "Este quadro de missões é destinado a aventureiros dos níveis 41 a 55.";
			close;
		}

		mes "[Quadro de Missões]";
		mes "Bem-vindo ao quadro de missões.";
		next;

		if (select("Consultar as missões.:Cancelar.") == 2) {
			mes "^000077Tudo bem. Vamos consultar as missões em outra ocasião.^000000";
			close;
		}

		switch(select("Caça aos Sapos de Thara:Caça aos Cruzadores:Caça aos Kukres:Caça aos Bebês Orc:Caça aos Guerreiros Orc:Caça aos Vadons:Caça aos Megalodons:Caça aos Marses:Caça aos Cornutus:Caça aos Mysts:Caça aos Shellfish:Caça às Esferas Marinhas:Caça aos Phens:Caça aos Esqueletos Orc:Caça aos Zenorcs:Caça às Múmias")) {

			case 1:
				callsub L_Quest,12072,"Você já viu um sapo vermelho? Ele se chama Sapo de Thara e é muito raro. Estamos tentando confirmar onde ele vive. Por favor, cace 20 Sapos de Thara.";

			case 2:
				callsub L_Quest,12073,"Ouvi dizer que alguns brinquedos estão atacando pessoas. Isso é um absurdo! Vamos mostrar nossa força! Precisamos caçar 15 Cruzadores!";

			case 3:
				callsub L_Quest,12074,"Alguém foi roubado perto da praia! O possível suspeito é um monstro criminoso chamado Kukre. Por favor, cace 30 Kukres.";

			case 4:
				callsub L_Quest,12075,"Os Bebês Orc estão ficando mais violentos ultimamente. Por favor, cace 15 Bebês Orc.";

			case 5:
				callsub L_Quest,12076,"Os Guerreiros Orc estão cada vez mais agressivos. Por favor, cace 20 Guerreiros Orc.";

			case 6:
				callsub L_Quest,12077,"Você já viu caranguejos com cascos grossos? Eles são chamados de Vadons. Por favor, cace 15 Vadons.";

			case 7:
				callsub L_Quest,12078,"Há peixes mortos por toda a praia, e a situação está piorando. Por favor, cace 30 Megalodons.";

			case 8:
				callsub L_Quest,12079,"Você pode caçar Marses? Preciso de 15 Marses para minha pesquisa. Por favor, ajude-me.";

			case 9:
				callsub L_Quest,12080,"Um Cornutus idiota mordeu o pé do prefeito! Ele está furioso, então precisamos caçar 20 Cornutus para deixá-lo feliz!";

			case 10:
				callsub L_Quest,12081,"As pessoas estão apavoradas por causa dos Myst. Eles parecem fantasmas e assustam todo mundo. Por favor, cace 15 Mysts.";

			case 11:
				callsub L_Quest,12082,"Não pergunte por quê, mas cace 30 Shellfish, por favor. É uma missão secreta!";

			case 12:
				callsub L_Quest,12083,"Esfera Marinha? Você sabe o que é uma Esfera Marinha? Tenho uma missão para você: cace 15 Esferas Marinhas.";

			case 13:
				callsub L_Quest,12084,"O prefeito quer comer um delicioso prato de peixe. Phen será o ingrediente perfeito. Por favor, cace 20 Phens!";

			case 14:
				callsub L_Quest,12085,"Os monstros Orc mortos voltaram à nossa cidade e ressuscitaram como Esqueletos Orc. Estamos em caos! Por favor, cace 15 Esqueletos Orc.";

			case 15:
				callsub L_Quest,12086,"Alguns monstros estranhos chamados Zenorc estão nos atacando. Por favor, cace 30 Zenorcs.";

			case 16:
				callsub L_Quest,12087,"Prepare-se! Esta é uma missão cruel! Cace 15 Múmias! Tentei uma vez, mas não consegui. É assustador demais!";
		}
		end;
	}

	setarray .@Exp[0],
		9600,7200,15750,7875,11100,8550,17100,9000,
		12300,9900,20250,10350,13800,11025,22500,11250;

	// Recompensa de moedas de acordo com a dificuldade da missão.
	// Valores baseados na quantidade de EXP da missão.
	setarray .@Coins[0],
		3,2,4,2,3,2,4,2,
		4,3,5,3,4,3,5,3;

	for (set .@i,0; .@i < 16; set .@i,.@i+1) {
		if (checkquest(12072 + .@i,HUNTING) == 2) {
			mes "[Quadro de Missões]";
			mes "Você concluiu a missão!";
			mes "Aqui está sua recompensa.";
			erasequest 12088;
			erasequest 12072 + .@i;
			specialeffect2 EF_STEAL;
			getexp .@Exp[.@i],0;

			// Entrega moedas ao concluir a missão.
			callfunc("F_EdenMissionCoinReward",.@Coins[.@i]);

			close;
		}
	}

	if ((checkquest(12072,HUNTING) == 0) &&
	    (checkquest(12073,HUNTING) == 0) &&
	    (checkquest(12074,HUNTING) == 0) &&
	    (checkquest(12075,HUNTING) == 0) &&
	    (checkquest(12076,HUNTING) == 0) &&
	    (checkquest(12077,HUNTING) == 0) &&
	    (checkquest(12078,HUNTING) == 0) &&
	    (checkquest(12079,HUNTING) == 0) &&
	    (checkquest(12080,HUNTING) == 0) &&
	    (checkquest(12081,HUNTING) == 0) &&
	    (checkquest(12082,HUNTING) == 0) &&
	    (checkquest(12083,HUNTING) == 0) &&
	    (checkquest(12084,HUNTING) == 0) &&
	    (checkquest(12085,HUNTING) == 0) &&
	    (checkquest(12086,HUNTING) == 0) &&
	    (checkquest(12087,HUNTING) == 0)) {

		mes "Agora você pode consultar novamente o quadro de missões.";
		erasequest 12088;
		close;
	}

	mes "Não há nada especial a fazer no momento.";
	close;

L_Quest:
	mes "[Descrição da Missão]";
	mes getarg(1);
	next;

	if (select("Aceitar a missão.:Cancelar.") == 1) {
		mes "[Quadro de Missões]";
		mes "Você aceitou a missão!";
		mes "Para consultar os detalhes, pressione Alt + U e abra a Janela de Missões.";
		setquest 12088;
		setquest getarg(0);
	}

	close;
}
