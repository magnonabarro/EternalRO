function	script	VIP_iRO_Acolyte	{
	if (!vip_status(VIP_STATUS_ACTIVE)) {
		mes getarg(0);
		mes "Porém, meus poderes completos só estão disponíveis para quem tem status VIP.";
		mes "Ainda posso oferecer um teletransporte para "+getarg(1)+" mesmo assim.";
		next;
		switch(select(getarg(0)+" -> 1800z:Cancelar")) {
		case 1:
			if (Zeny < 1800) {
				mes getarg(0);
				mes "Sinto muito, mas você não tem";
				mes "zenys suficientes para o serviço";
				mes "de teletransporte. A taxa para ir";
				mes "para "+getarg(1)+" custa 1.800 zenys.";
				close;
			}
			specialeffect2 EF_READYPORTAL;
			specialeffect2 EF_TELEPORTATION;
			specialeffect2 EF_PORTAL;
			next;
			set Zeny, Zeny - 1800;
			if (getarg(1) == "Juno")
				warp "yuno",158,125;
			else
				warp "prontera",116,72;
			close;
		case 2:
			close;
		}
	}
	return;
}

	next;
	if(select("Sim, pode me entregar!:Não, espere um pouco.") == 2)
		close;

	set .@total_amount, getarg(3) * .@amount;

		mes "[Máquina de Trocas]";
		mes "Você não tem "+getitemname(getarg(0))+" suficiente para realizar essa troca.";
