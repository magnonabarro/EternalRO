S_Quest:
	if (isbegin_quest(getarg(0)))
		erasequest getarg(0);
	setquest getarg(1);
	if (getarg(2,0)) {
		getexp getarg(2),getarg(3);
		callfunc("F_EdenMissionCoinReward",9);
	}
	return;
S_Hunting:
	.@total = getargcount();
	for ( .@i = 0; .@i < .@total; .@i += 3 ) {
		.@quest_id = getarg(.@i);
		switch( checkquest(.@quest_id,HUNTING) ) {
		case -1:
			if (checkquest(.@quest_id+1) > -1)
				erasequest (.@quest_id+1);
			continue;
		case 2:
			mes "[Mingmin]";
			mes "Voce concluiu a pesquisa sobre "+ getarg(.@i+2) +". Obrigado.";
			getexp 700000,getarg(.@i+1);
			callfunc("F_EdenMissionCoinReward",9);
			break;
		case 0:
		case 1:
			mes "[Mingmin]";
			mes "Vou remover "+ getarg(.@i+2) +" da lista de solicitacoes de pesquisa.";
		}
		erasequest .@quest_id;
		next;
	}
	return;
S_Gathering:
	.@quest_id = getarg(0);
	if (isbegin_quest(.@quest_id) > 0) {
		if (getarg(5,0))
			.@count_item2 = countitem(getarg(5));
		mes "[Mingmin]";
		if (countitem(getarg(3)) >= getarg(4) && .@count_item2 >= getarg(6,0)) {
			if (.@count_item2)
				mes "Voce concluiu a pesquisa sobre "+ getitemname(getarg(3)) +" e "+ getitemname(getarg(5)) +". Obrigado.";
			else
				mes "Voce concluiu a pesquisa sobre "+ getitemname(getarg(3)) +". Obrigado.";
			delitem getarg(3),getarg(4);
			if (getarg(5,0))
				delitem getarg(5),getarg(6);
			getexp 700000,getarg(1);
			callfunc("F_EdenMissionCoinReward",9);
		}
		else {
			if (getarg(5,0))
				mes "Vou remover "+  getitemname(getarg(3)) +" e "+ getitemname(getarg(5)) +" da lista de solicitacoes de pesquisa.";
			else
				mes "Vou remover "+  getitemname(getarg(3)) +" da lista de solicitacoes de pesquisa.";
		}
		erasequest .@quest_id;
		next;
	}
	else if (isbegin_quest(.@quest_id+1) > 0)
		erasequest (.@quest_id+1);
	return;
