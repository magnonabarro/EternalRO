L_Quest:
	set .@quest1, getarg(0);
	set .@quest2, .@quest1+31;
	.@cutin$ = getarg(3);
	set .@playtime, checkquest(.@quest2,PLAYTIME);
	if (.@playtime == 0 || .@playtime == 1) {
		mes "[Quadro de Missões 86-90]";
		mes "Você precisa esperar 3 horas antes de aceitar esta missão novamente.";
		close;
	}
	set .@hunting, checkquest(.@quest1,HUNTING);
	if (.@hunting == 0 || .@hunting == 1) {
		mes "[Quadro de Missões 86-90]";
		mes "Você já terminou de caçar os 30 monstros da espécie "+getarg(1)+"?";
		close;
	} else if (.@hunting == 2) {
		mes "[Quadro de Missões 86-90]";
		mes "Você concluiu a caçada.";
		mes "Aqui está sua recompensa pela conclusão da missão.";
		erasequest .@quest1;
		if (.@playtime > -1) erasequest .@quest2;
		setquest .@quest2;
		getexp getarg(2),0;
		callsub L_RewardCoins,.@quest1;
		close;
	}/* else if (.@hunting == 3) {
//		recall_completequest .@quest1

		mes "Você aceitou novamente a missão de caça: "+getarg(1)+".";
		close;

		mes "Derrote 30 monstros da espécie "+getarg(1)+" nessa região e volte a este quadro para registrar a conclusão.";

			mes "Quando terminar, volte a este quadro para registrar a conclusão e receber sua recompensa.";
