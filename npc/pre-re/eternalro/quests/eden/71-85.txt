moc_para01,44,38,3	script	Mission [71 - 85]	857,{
	if (!callfunc("F_HasEdenGroupMark")) {
		mes "- Somente membros do -";
		mes "- Grupo Éden podem consultar -";
		mes "- este quadro de missões. -";
		close;
	}

	// Limpa os dados das missões antigas, que não são mais utilizadas.
	if (slv_quest) set slv_quest,0;

	if (checkquest(10102) > -1 || checkquest(10103) > -1 ||
	    checkquest(10104) > -1 || checkquest(10105) > -1 ||
	    checkquest(10106) > -1) {

		for(set .@quest,10102; .@quest<=10106; set .@quest,.@quest+1) {
			if (checkquest(.@quest) > -1)
				erasequest .@quest;
		}

		mes "- Como o cliente foi transferido -";
		mes "- para outro local, -";
		mes "- o formulário de inscrição foi removido. -";
		next;
	}

	if (BaseLevel < 71) {
		mes "- Somente personagens -";
		mes "- do nível adequado podem -";
		mes "- consultar este quadro. -";
		mes "- Procure um quadro de missões -";
		mes "- adequado ao seu nível. -";
		close;
	}
	else if (BaseLevel > 85) {

		// IDs das missões: 10107-10123 e 5055-5057.
		setarray .@quests[0],
			10107,10108,10109,
			10110,10111,10112,10113,
			10114,10115,10116,10117,
			10118,10119,10120,
			10121,10122,10123,5055,5056;

		for(set .@i,0; .@i<getarraysize(.@quests); set .@i,.@i+1) {
			if (checkquest(.@quests[.@i],HUNTING) == 2)
				set .@complete[getarraysize(.@complete)],.@quests[.@i];
		}

		if (checkquest(5057) > -1 && countitem(7187) >= 30)
			set .@complete[getarraysize(.@complete)],5057;

		if (getarraysize(.@complete)) {
			mes "Você possui missões concluídas.";
			mes "Deseja entregar as missões e receber suas recompensas?";
			next;

			if(select("Sim, entregar as missões.:Não, cancelar.") == 2)
				close;

			for(set .@i,0; .@i<getarraysize(.@complete); set .@i,.@i+1)
				callsub L_Quest,.@complete[.@i];

			close;
		}

		mes "- Somente personagens -";
		mes "- do nível adequado podem -";
		mes "- consultar este quadro. -";
		mes "- Procure um quadro de missões -";
		mes "- adequado ao seu nível. -";
		close;
	}

	mes "- Escolha a região -";
	mes "- onde deseja caçar. -";
	next;

	switch(select("Masmorra das Esfinges:Abadia de Glast Heim:Região de Juno:Torre do Relógio:Regiões especiais")) {

	case 1:
		mes "- Este quadro possui várias -";
		mes "- missões relacionadas à -";
		mes "- Masmorra das Esfinges. -";
		next;

		set .@str1$,"Os moradores de Morroc estão preocupados com o número crescente de monstros na Masmorra das Esfinges. ";
		set .@str2$,"- Meidi, de Morroc -";

		switch(select("Caçar Réquiens:Caçar Marduks:Caçar Pasanas")) {
			case 1:
				callsub L_Quest,10107,.@str1$+"Cace 10 Réquiens.",.@str2$;

			case 2:
				callsub L_Quest,10108,.@str1$+"Cace 10 Marduks.",.@str2$;

			case 3:
				callsub L_Quest,10109,.@str1$+"Cace 10 Pasanas.",.@str2$;
		}

	case 2:
		mes "- Este quadro possui várias -";
		mes "- missões relacionadas à -";
		mes "- região de Glast Heim. -";
		next;

		set .@str1$,"Tenho muito medo de que os monstros de Glast Heim venham nos atacar. ";
		set .@str2$,"- Ancellia, de Geffen -";

		switch(select("Caçar Molduras Negras:Caçar Druidas Malignos:Caçar Espectros:Caçar Arqueiros Raydric")) {
			case 1:
				callsub L_Quest,10110,.@str1$+"Cace 10 Molduras Negras.",.@str2$;

			case 2:
				callsub L_Quest,10111,.@str1$+"Cace 10 Druidas Malignos.",.@str2$;

			case 3:
				callsub L_Quest,10112,.@str1$+"Cace 10 Espectros.",.@str2$;

			case 4:
				callsub L_Quest,10113,.@str1$+"Cace 10 Arqueiros Raydric.",.@str2$;
		}

	case 3:
		mes "- Este quadro possui várias -";
		mes "- missões relacionadas à -";
		mes "- região ao redor de Juno. -";
		next;

		set .@str1$,"Para pessoas que não sabem lutar, como eu, viajar está cada vez mais difícil devido à grande quantidade de monstros. ";
		set .@str2$,"- Jeanbai, viajante -";

		switch(select("Caçar Grand Pecos:Caçar Sleepers:Caçar Bodes:Caçar Harpias")) {
			case 1:
				callsub L_Quest,10114,.@str1$+"Cace 20 Grand Pecos.",.@str2$;

			case 2:
				callsub L_Quest,10115,.@str1$+"Cace 20 Sleepers.",.@str2$;

			case 3:
				callsub L_Quest,10116,.@str1$+"Cace 20 Bodes.",.@str2$;

			case 4:
				callsub L_Quest,10117,.@str1$+"Cace 20 Harpias.",.@str2$;
		}

	case 4:
		mes "- Este quadro possui várias -";
		mes "- missões relacionadas à -";
		mes "- Torre do Relógio. -";
		next;

		set .@str1$,"Eu costumava adorar ouvir a doce melodia da Torre do Relógio de Al De Baran, mas os ruídos dos monstros não me deixam mais escutá-la. ";
		set .@str2$,"- Rizingsetter, presidente da comunidade Amor ao Relógio -";

		switch(select("Caçar Clocks:Caçar Punks:Caçar Ridewords")) {
			case 1:
				callsub L_Quest,10118,.@str1$+"Cace 15 Clocks antes que o relógio pare de funcionar.",.@str2$;

			case 2:
				callsub L_Quest,10119,.@str1$+"Cace 15 Punks antes que o relógio pare de funcionar.",.@str2$;

			case 3:
				callsub L_Quest,10120,.@str1$+"Cace 15 Ridewords antes que o relógio pare de funcionar.",.@str2$;
		}

	case 5:
		mes "- Este quadro possui várias -";
		mes "- missões relacionadas a -";
		mes "- regiões especiais. -";
		next;

		set .@str1$,"Quero sair em uma aventura, mas tenho tantas coisas para resolver que não consigo. Se eu não consigo, os outros também não conseguirão! ";
		set .@str2$,"- Funfy, que deseja sair em uma aventura -";

		switch(select("Caçar Kikimoras:Caçar Bonecas Miyabi:Caçar Mi Gaos:Caçar Mulas sem Cabeça:Caçar Tamruans:Recolher Máscaras do Festival")) {

			case 1:
				callsub L_Quest,10121,.@str1$+"Cace 15 Kikimoras em Moscovia!",.@str2$;

			case 2:
				callsub L_Quest,10122,.@str1$+"Cace 15 Bonecas Miyabi em Amatsu!",.@str2$;

			case 3:
				callsub L_Quest,10123,.@str1$+"Cace 15 Mi Gaos em Louyang!",.@str2$;

			case 4:
				callsub L_Quest,5055,"Sei que existem muitos monstros assustadores fora da cidade, mas o que realmente me apavora são as Mulas sem Cabeça que ficam rondando por aqui. Espero que nunca entrem na cidade. Cace 15 Mulas sem Cabeça!","- Mariana, de Brasilis -";

			case 5:
				callsub L_Quest,5056,"Os Tamruans fazem muito barulho. Não me importo durante o dia, mas eles não me deixam dormir à noite. Quero dormir! Cace 15 Tamruans!","- Xamã Pumo, de Ayothaya -";

			case 6:
				callsub L_Quest,5057,"As Crianças de Bolinho roubaram todas as máscaras que preparei para o festival. Tente recuperar 30 máscaras! Ficarei em dívida com você~","- Chefe SaYumMoon, de Kunlun -";
		}
	}

	end;


// callsub L_Quest, <ID da missão> {, "<descrição 1>", "<descrição 2>"};

L_Quest:
	set .@quest,getarg(0);

	switch(.@quest) {
		case 10107: set .@name$,"Caçar Réquiens"; set .@exp,15000; break;
		case 10108: set .@name$,"Caçar Marduks"; set .@exp,15000; break;
		case 10109: set .@name$,"Caçar Pasanas"; set .@exp,15000; break;

		case 10110: set .@name$,"Caçar Molduras Negras"; set .@exp,20000; break;
		case 10111: set .@name$,"Caçar Druidas Malignos"; set .@exp,20000; break;
		case 10112: set .@name$,"Caçar Espectros"; set .@exp,20000; break;
		case 10113: set .@name$,"Caçar Arqueiros Raydric"; set .@exp,20000; break;

		case 10114: set .@name$,"Caçar Grand Pecos"; set .@exp,30000; break;
		case 10115: set .@name$,"Caçar Sleepers"; set .@exp,30000; break;
		case 10116: set .@name$,"Caçar Bodes"; set .@exp,30000; break;
		case 10117: set .@name$,"Caçar Harpias"; set .@exp,30000; break;

		case 10118: set .@name$,"Caçar Clocks"; set .@exp,25000; break;
		case 10119: set .@name$,"Caçar Punks"; set .@exp,25000; break;
		case 10120: set .@name$,"Caçar Ridewords"; set .@exp,25000; break;

		case 10121: set .@name$,"Caçar Kikimoras"; set .@exp,30000; break;
		case 10122: set .@name$,"Caçar Bonecas Miyabi"; set .@exp,30000; break;
		case 10123: set .@name$,"Caçar Mi Gaos"; set .@exp,30000; break;

		case 5055: set .@name$,"Caçar Mulas sem Cabeça"; set .@exp,30000; break;
		case 5056: set .@name$,"Caçar Tamruans"; set .@exp,30000; break;
		case 5057: set .@name$,"Recolher Máscaras do Festival"; set .@exp,30000; break;
	}

	// Recompensas em moedas conforme a dificuldade da missão.
	switch(.@quest) {
		case 10107:
		case 10108:
		case 10109:
			set .@coins,4;
			break;

		case 10110:
		case 10111:
		case 10112:
		case 10113:
		case 10118:
		case 10119:
		case 10120:
			set .@coins,5;
			break;

		default:
			set .@coins,6;
			break;
	}

	// Entrega de missão já concluída por personagem acima do nível permitido.
	if (getargcount() == 1) {
		mes "Você concluiu a missão '"+.@name$+"'.";
		mes "Deseja entregá-la e receber suas recompensas?";
		next;

		if(select("Entregar a missão.:Cancelar.") == 1) {
			if (.@quest == 5057)
				delitem 7187,30; // Máscara do Festival

			erasequest .@quest;
			getexp .@exp,0;

			// Entrega moedas ao concluir a missão.
			callfunc("F_EdenMissionCoinReward",.@coins);

			mes "Você recebeu experiência e moedas pela conclusão da missão.";
			next;
			return;
		}
		close;
	}
	else {
		if (.@quest == 5057) {
			// A missão 5057 é a única missão de coleta.
			if (checkquest(5057) == -1)
				set .@hunting,-1;
			else if (countitem(7187) >= 30)
				set .@hunting,2;
		}
		else
			set .@hunting,checkquest(.@quest,HUNTING);

		if (.@hunting == 0 || .@hunting == 1) {
			mes getarg(1);
			mes getarg(2);
			next;

			mes "Volte aqui quando terminar a missão.";
		}
		else if (.@hunting == 2) {
			mes "Você concluiu a missão.";
			next;

			if(select("Entregar a missão.:Cancelar.") == 1) {
				if (.@quest == 5057)
					delitem 7187,30; // Máscara do Festival

				erasequest .@quest;
				getexp .@exp,0;

				// Entrega moedas ao concluir a missão.
				callfunc("F_EdenMissionCoinReward",.@coins);

				mes "Você recebeu experiência e moedas pela conclusão da missão.";
			}
		}
		/*
		else if (.@hunting == 3) {
			erasequest .@quest;
			mes "Você aceitou novamente a missão '"+.@name$+"'.";
		}
		*/
		else {
			mes getarg(1);
			mes getarg(2);
			next;

			mes "Deseja aceitar esta missão?";
			next;

			if(select("Sim, aceitar.:Não, cancelar.") == 1) {
				mes "Você aceitou a missão '"+.@name$+"'.";
				mes "Consulte a Janela de Missões para ver mais detalhes.";
				setquest .@quest;
			}
		}

		close;
	}
}
