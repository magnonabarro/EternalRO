S_Quest:
	if (isbegin_quest(getarg(0)))
		erasequest getarg(0);
	setquest getarg(1);
	if (getarg(2,0)) {
		getexp getarg(2),getarg(3);
		// Recompensa por concluir a missão: os monstros de Lighthalzen valem 11 moedas,
		// as Encarnações de Morroc valem 12 e os Scarabas valem 13.
		if (getarg(1) >= 13144 && getarg(1) <= 13154)
			callfunc("F_EdenMissionCoinReward",11);
		else if (getarg(1) == 13156 || getarg(1) == 13158 || getarg(1) == 13160)
			callfunc("F_EdenMissionCoinReward",12);
		else
			callfunc("F_EdenMissionCoinReward",13);
	}
	return;
S_Erasequest:
	.@total = getargcount();
	for ( .@i = 0; .@i < .@total; .@i += 3 ) {
		.@quest_id = getarg(.@i);
		switch( checkquest(.@quest_id,HUNTING) ) {
		case -1:
			if (checkquest(.@quest_id+1) > -1)
				erasequest (.@quest_id+1);
			continue;
		case 0:
		case 1:
			mes "[Ragi]";
			switch( getarg(.@i+1) ) {
			case 1:
				mes "Tudo bem quanto ao restante da história de "+ getarg(.@i+2) +". Vou perguntar a outra pessoa.";
				break;
			case 2:
			case 3:
				mes "Vou perguntar a outros aventureiros sobre a história de "+ getarg(.@i+2) +".";
				break;
			}
			break;
		case 2:
			mes "[Ragi]";
			switch( getarg(.@i+1) ) {
			case 1:
				mes "Você descobriu a história de "+ getarg(.@i+2) +"? Obrigado! Prometo transformá-la em uma bela canção.";
				getexp 1100000,300000;
				callfunc("F_EdenMissionCoinReward",11);
				break;
			case 2:
				mes "A história de Morroc será boa para uma canção? Você fez um ótimo trabalho.";
				getexp 1100000,200000;
				callfunc("F_EdenMissionCoinReward",12);
				break;
			case 3:
				mes "Como foi sua aventura? Foi divertida? Um dia, conte-me sua história.";
				getexp 1100000,200000;
				callfunc("F_EdenMissionCoinReward",13);
				break;
			}
		}
		erasequest .@quest_id;
		next;
	}
	return;
