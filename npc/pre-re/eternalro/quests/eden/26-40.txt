moc_para01,38,38,3	script	Mission [26 - 40]	857,{
	if (!callfunc("F_HasEdenGroupMark")) {
		mes "Você não tem autorização para consultar este quadro de missões.";
		close;
	}

	mes "[Quadro de Missões - Níveis 26 a 40]";
	mes "- Há várias missões -";
	mes "- disponíveis neste quadro. -";
	next;

	if ((BaseLevel < 26) || (BaseLevel > 40)) {
		mes "Seu nível não é adequado para consultar este quadro de missões.";
		close;
	}

	switch(select("Missão de caça:Missão de entrega:Missão de coleta:Missão de escolta")) {
	case 1:
		mes "[Quadro de Missões de Caça]";
		mes "- Os detalhes das missões de caça -";
		mes "- estão listados aqui. -";
		next;

		switch(select("Caçar Esporos Venenosos:Caçar Smokies:Caçar Salgueiros Anciões (Nível 30+):Caçar Cocos (Nível 35+):Caçar Sapos Thara (Nível 38+)")) {
			case 1: callsub L_Quest,8266,0,"Caçar Esporos Venenosos","Sponiac","Geffen","Cace 10 Esporos Venenosos.";
			case 2: callsub L_Quest,8267,0,"Caçar Smokies","Sponiac","Geffen","Cace 10 Smokies.";
			case 3: callsub L_Quest,8268,30,"Caçar Salgueiros Anciões","Sponiac","Geffen","Cace 10 Salgueiros Anciões.";
			case 4: callsub L_Quest,8269,35,"Caçar Cocos","Sponiac","Geffen","Cace 10 Cocos.";
			case 5: callsub L_Quest,8270,38,"Caçar Sapos Thara","Sponiac","Geffen","Cace 10 Sapos Thara.";
		}

	case 2:
		mes "[Quadro de Missões de Entrega]";
		mes "- Os detalhes das missões de entrega -";
		mes "- estão listados aqui. -";
		next;

		switch(select("Entregar um inseticida:Entregar mel:Entregar cobertores (Nível 35+)")) {
			case 1: callsub L_Quest,8271,0,"Entregar um inseticida","Mensageiro Dieshin","Ruínas de Morroc","Pegue um 'Inseticida' com Dieshin e entregue-o a sudeste de Payon.";
			case 2: callsub L_Quest,8272,0,"Entregar mel","Mensageiro Dieshin","Ruínas de Morroc","Pegue 'Mel' com Dieshin e entregue-o a oeste da Abadia de Santa Capitolina.";
			case 3: callsub L_Quest,8273,35,"Entregar cobertores","Mensageiro Dieshin","Ruínas de Morroc","Pegue os cobertores com Dieshin e entregue-os no campo ao sul de Lutie.";
		}

	case 3:
		mes "[Quadro de Missões de Coleta]";
		mes "- Os detalhes das missões de coleta -";
		mes "- estão listados aqui. -";
		next;

		switch(select("Coletar ossos:Coletar peles de animais:Coletar cascas (Nível 30+):Coletar caudas (Nível 32+):Coletar biscoitos (Nível 35+):Coletar bigodes (Nível 35+)")) {
			case 1: callsub L_Quest,8274,0,"Coletar ossos","Dashia","Payon","Colete 10 'Ossos de Esqueleto' e entregue-os a Dashia.";
			case 2: callsub L_Quest,8275,0,"Coletar peles de animais","Dashia","Payon","Colete 10 'Pés de Urso' e 5 'Peles de Animal' e entregue-os a Dashia.";
			case 3: callsub L_Quest,8276,30,"Coletar cascas","Dashia","Payon","Colete 20 'Cascas de Verme' e 20 'Garlets' e entregue-os a Dashia.";
			case 4: callsub L_Quest,8277,32,"Coletar caudas","Dashia","Payon","Colete 10 'Caudas de Yoyo', 10 'Bolotas' e 10 'Folhas de Racoon' e entregue-as a Dashia.";
			case 5: callsub L_Quest,8278,35,"Coletar biscoitos","Dashia","Payon","Colete 5 'Biscoitos Bem-Assados' e 1 'Pedaço de Bolo' e entregue-os a Dashia.";
			case 6: callsub L_Quest,8279,35,"Coletar bigodes","Dashia","Payon","Colete 5 'Bigodes de Toupeira' e 3 'Garras de Toupeira' e entregue-as a Dashia.";
		}

	case 4:
		mes "[Quadro de Missões de Escolta]";
		mes "Não há missões de escolta cadastradas no quadro no momento.";
		close;
	}
	end;

L_Quest:
	// ID da missão, nível necessário, missão, solicitante, local, descrição
	.@string$ = getarg(2);

	if (BaseLevel < getarg(1,26)) {
		mes "Seu nível não atende aos requisitos. Você não pode aceitar a missão '"+ .@string$ +"'.";
		close;
	}

	mes "["+ .@string$ +" !!]";
	mes "Solicitante: "+getarg(3);
	mes "Local do solicitante: "+getarg(4);
	mes "Missão solicitada: "+getarg(5);
	mes " ";
	mes "Você aceita esta missão?";
	next;

	if(select("Não, não vou aceitar.:Sim, vou aceitar.") == 1) {
		mes "Você recusou a missão '"+ .@string$ +"'.";
		close;
	}

	if (checkquest(getarg(0)) > -1)
		mes "Você já está realizando esta missão.";
	else {
		setquest getarg(0);
		mes "Você aceitou a missão '"+ .@string$ +"'.";
	}

	mes "Você receberá uma recompensa do solicitante após concluir a missão.";
	next;
	mes "Para ver mais detalhes, pressione Alt + U e consulte as informações na Janela de Missões.";
	close;
}
