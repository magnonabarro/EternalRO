function	script	missevent	{

		}
		set missaoativa,1;
	}

		mes "Você deve me trazer 50x "+getitemname(7156)+".";
		mes "Acha que consegue?";

			mes "Certo.";
			close;
		}
		set missaoativa,2;		
	}
	if(getarg(0) == 3){
		mes "Você deve me trazer 40x "+getitemname(7099)+".";
		mes "Acha que consegue?";
		if(select("Sim:Não")==2){
			mes "Certo.";
			close;
		}
		set missaoativa,3;
	}
	if(getarg(0) == 4){
		mes "Você deve matar 20x "+getmonsterinfo(1499,0)+".";
		mes "Acha que consegue?";
		if(select("Sim:Não")==2){

function	script	missprem	{
