L_Offer:
	.@qstatus = checkquest(.@qid,HUNTING);
	
	if (.@qstatus == 2) {
		// Quest completa - entregar
		mes "[Quadro Missões Especiais]";
		mes "Você já completou esta missão. Entregue ao Monstrólogo para registrar.";
		next;
		goto L_ShowMenu;
	}
	
	if (.@qstatus != -1) {
		// Quest já em andamento
		mes "[Quadro Missões Especiais]";
		mes "Você já está em uma missão semelhante.";
		next;
		goto L_ShowMenu;
	}
	
	// Quest disponível - oferecer para pegar
	if (BaseLevel < .@min) {
		mes "[Quadro Missões Especiais]";
		mes "Esta missão é muito desafiadora para você no momento.";
		next;
		goto L_ShowMenu;
	}
	
	mes "[Quadro Missões Especiais]";
	mes "Deseja aceitar esta missão?";
	next;
	if(select("Aceitar:Negar") == 1) {
		setquest .@qid;
		mes "Boa sorte! Encontre e derrote o monstro.";
		next;
		goto L_ShowMenu;
	} else {
		goto L_ShowMenu;
	}
