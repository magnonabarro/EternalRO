treasure01,25,31,4	script	Caixa Antiga#1	844,{
    if (countitem(25074) == 0) { mes "[Caixa Antiga]"; mes "Vazia."; close; }
    mes "- Esconder dobrões aqui? -";
    next;
    if (select("Esconder rapidamente.", "Acho que tem alguém olhando.") == 2) { close; }
    close2;
    if (progressbar(1500, 1)) {
        if (countitem(25074) > 0) {
            delitem 25074, 1;
            npctalk "Tire suas mãos sujas daqui pirata!";
        }
    }
    donpcevent strnpcinfo(0) + "::OnCooldown";
    end;
OnCooldown:
    hideonnpc strnpcinfo(0);
    sleep 300000;
    hideoffnpc strnpcinfo(0);
    end;
}
