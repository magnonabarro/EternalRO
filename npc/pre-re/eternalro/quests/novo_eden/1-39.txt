	}

L_Handle:
	.@qstatus = checkquest(.@qid,HUNTING);
	
	if (.@qstatus == 2) {
		// Quest completa - entregar
		// Verificar itens apenas se a quest exigir (i1 > 0 ou i2 > 0)
		if (.@i1 > 0) {
			if (countitem(.@i1) < .@a1) {
				mes "Você não tem os itens necessários para esta missão.";
				mes "Precisa de " + .@a1 + " " + getitemname(.@i1) + ".";
				next;
				goto L_ShowMenu;
			}
		}
		if (.@i2 > 0) {
			if (countitem(.@i2) < .@a2) {
				mes "Você não tem os itens necessários para esta missão.";
				mes "Precisa de " + .@a2 + " " + getitemname(.@i2) + ".";
				next;
				goto L_ShowMenu;
			}
		}
		// Remover itens se necessário
		if (.@i1 > 0) delitem .@i1,.@a1;
		if (.@i2 > 0) delitem .@i2,.@a2;
		
		getexp .@exp,.@job_exp;
		erasequest .@qid;
		setquest .@cooldown_qid;
		mes "Missão entregue! Recompensa: " + .@exp + " EXP de Base e " + .@job_exp + " EXP de Classe.";
		next;
		goto L_ShowMenu;
	}
	
	if (.@qstatus != -1) {
		// Quest já em andamento
		mes "Você já tem esta missão em andamento.";
		next;
		goto L_ShowMenu;
	}
	
	// Verificar se o cooldown está ativo
	if (checkquest(.@cooldown_qid, HUNTING) != -1) {
		mes "Você já completou esta missão recentemente. Aguarde o tempo de espera para pegá-la novamente.";
		next;
		goto L_ShowMenu;
	}
	
	// Quest disponível - oferecer para pegar
	if (BaseLevel < .@min) {
		mes "Esta missão requer nível mínimo de " + .@min + ". Devo procurar missões mais fáceis.";
		next;
		goto L_ShowMenu;
	}
	
	// Mostrar objetivo da missão
	mes "Objetivo da Missão:";
	if (.@qid == 99001) mes "Caçar 5 [Salgueiros] + Coletar 3 [Tronco Estéril]";
	else if (.@qid == 99002) mes "Caçar 5 [Jiboia]";
	else if (.@qid == 99003) mes "Caçar 5 [Rabo de Verme]";
	else if (.@qid == 99004) mes "Caçar 10 [Esporo]";
	else if (.@qid == 99005) mes "Caçar 10 [Zangão]";
	else if (.@qid == 99006) mes "Caçar 10 [Rocker]";
	else if (.@qid == 99007) mes "Caçar 10 [Ambernite]";
	else if (.@qid == 99008) mes "Caçar 10 [Zumbi] + 20 [Esqueleto]";
	else if (.@qid == 99009) mes "Caçar 10 [Muka] + 20 [PecoPeco]";
	else if (.@qid == 99010) mes "Caçar 10 [Sapo de Thara] + Coletar 10 [Pata Pegajosa]";
	else if (.@qid == 99011) mes "Caçar 10 [Hidra] + Coletar 5 [Tentáculo]";
	else if (.@qid == 99012) mes "Caçar 10 [Megalodon]";
	next;
	
	mes "Deseja aceitar a missão?";
	next;
	if(select("Aceitar:Negar") == 1) {
		setquest .@qid;
		mes "Missão aceita!";
		next;
		goto L_ShowMenu;
	} else {
		goto L_ShowMenu;
	}
}
