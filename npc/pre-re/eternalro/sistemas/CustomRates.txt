				next;
				continue;

					mes "EXP: 1x | Drops: +50% | Descontos: 60%.";
					mes "Agente VIP sem ser VIP e @buff sem ser VIP.";
					mes "Esses recursos exigem scripts adicionais para realmente funcionar.";
