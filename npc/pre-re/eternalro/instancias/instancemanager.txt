L_FindIndex:
	.@selected$ = @im_selected_name$;
	for (.@i = 0; .@i < getarraysize(.DisplayName$); .@i++) {
		if (.DisplayName$[.@i] == .@selected$)
			return .@i;
	}
	return -1;

L_CheckCooldown:
	.@idx = getarg(0);
	// Mantem o cooldown de 24 horas do Torneio de Geffen.
	if (.@idx == 5 && GeffenTournament > gettimetick(2)) { dispbottom "Torneio de Geffen: tempo de espera ativo."; return 0; }
	return 1;

L_ClearExpiredCooldown:
	.@idx = getarg(0);
	if (.CooldownQuest[.@idx] > 0 && checkquest(.CooldownQuest[.@idx], PLAYTIME) == 2)
		erasequest .CooldownQuest[.@idx];
	if (.AuxQuest[.@idx] > 0 && checkquest(.AuxQuest[.@idx], PLAYTIME) == 2)
		erasequest .AuxQuest[.@idx];
	return;

L_Create:
	.@idx = callsub(L_FindIndex);
	if (.@idx < 0) { dispbottom "Gerenciador: instância não cadastrada."; return; }
	if (callsub(L_CheckCooldown, .@idx) == 0) return;
	.@party_id = getcharid(1);
	if (.@party_id > 0) {
		.@owner_mode = IM_PARTY;
		.@owner_id = .@party_id;
	} else {
		.@owner_mode = IM_CHAR;
		.@owner_id = getcharid(0);
	}
	.@party_oldid = instance_id(IM_PARTY);
	.@char_oldid = instance_id(IM_CHAR);
	if (.@party_oldid > 0 && instance_live_info(ILI_NAME, .@party_oldid) == .InstanceName$[.@idx]) { dispbottom "A instância já foi criada. Clique em Entrar."; return; }
	if (.@char_oldid > 0 && instance_live_info(ILI_NAME, .@char_oldid) == .InstanceName$[.@idx]) { dispbottom "A instância já foi criada. Clique em Entrar."; return; }
	if (.@party_oldid > 0 || .@char_oldid > 0) {
		dispbottom "Voce ou seu grupo ja possui outra instancia ativa. Termine-a antes de criar outra.";
		return;
	}
	.@cd = -1;
	if (.CooldownQuest[.@idx] > 0) .@cd = checkquest(.CooldownQuest[.@idx], PLAYTIME);
	if (.@cd == 2) { callsub L_ClearExpiredCooldown, .@idx; .@cd = -1; }
	if (.@cd == 0 || .@cd == 1) { dispbottom "Gerenciador: tempo de espera de 24 horas ativo para esta instância."; return; }
	if (.@idx == 5) { for (.@q = 9284; .@q <= 9314; .@q++) if (checkquest(.@q) != -1) erasequest .@q; }
	.@created_id = instance_create(.InstanceName$[.@idx], .@owner_mode, .@owner_id);
	if (.@created_id < 0) { dispbottom "Gerenciador: falha ao criar a instância. Verifique o campo Name e os mapas em instance_db.yml."; return; }
	dispbottom "Instancia criada. Voce ainda nao foi teleportado; clique em Entrar quando estiver pronto.";
	return;

L_Enter:
	.@idx = callsub(L_FindIndex);
	if (.@idx < 0) { dispbottom "Gerenciador: instância não cadastrada."; return; }
	if (callsub(L_CheckCooldown, .@idx) == 0) return;
	.@iid = instance_id(IM_PARTY);
	if (.@iid <= 0 || instance_live_info(ILI_NAME, .@iid) != .InstanceName$[.@idx]) .@iid = instance_id(IM_CHAR);
	if (.@iid <= 0 || instance_live_info(ILI_NAME, .@iid) != .InstanceName$[.@idx]) {
		dispbottom "Voce ou seu grupo ainda nao criou esta instancia. Clique em Criar primeiro.";
		return;
	}
	.@cd = -1;
	if (.CooldownQuest[.@idx] > 0) .@cd = checkquest(.CooldownQuest[.@idx], PLAYTIME);
	if (.@cd == 2) { callsub L_ClearExpiredCooldown, .@idx; .@cd = -1; }
	if (.@cd == 0 || .@cd == 1) {
		if (.@idx == 0) {
			.@aux = checkquest(60201, PLAYTIME);
			if (.@aux < 0 || .@aux >= 2 || etower_partyid != getcharid(1)) { dispbottom "Torre Sem Fim: a reentrada só pode ser feita pelo mesmo grupo durante o período original de 4 horas."; return; }
		} else if (.AuxQuest[.@idx] > 0 && .@idx == 3) {
			.@aux = checkquest(.AuxQuest[.@idx], PLAYTIME); if (.@aux == 0 || .@aux == 1) { dispbottom "Ninho de Nidhogg: o período auxiliar da missão ainda está ativo."; return; }
		}
	}
	.@result = instance_enter(.InstanceName$[.@idx], -1, -1, getcharid(0), .@iid);
	if (.@result != IE_OK) {
		if (.@result == IE_NOINSTANCE) dispbottom "A instancia nao existe mais. Clique em Criar novamente.";
		else if (.@result == IE_NOMEMBER) dispbottom "Voce nao pertence ao grupo dono desta instancia.";
		else dispbottom "Falha ao entrar na instancia (codigo " + .@result + ").";
		return;
	}
	// Mantem quest flags oficiais usados pelo progresso; o cooldown de 24 horas permanece ativo.
	if (.CooldownQuest[.@idx] > 0 && checkquest(.CooldownQuest[.@idx]) == -1) setquest .CooldownQuest[.@idx];
	switch (.@idx) {
	case 13: if (checkquest(15050) == -1) setquest 15050; if (checkquest(15051) == -1) setquest 15051; break;
	case 14: if (checkquest(12377) == -1) setquest 12377; if (checkquest(12378) == -1) setquest 12378; break;
	case 16: if (checkquest(12279) == -1) setquest 12279; break;
	case 17: if (checkquest(12327) == -1) setquest 12327; if (checkquest(12328) == -1) setquest 12328; if (checkquest(12329) == -1) setquest 12329; break;
	case 18: if (checkquest(15007) == -1) setquest 15007; if (checkquest(15008) == -1) setquest 15008; break;
	case 21: if (checkquest(7568) == -1) setquest 7568; break;
	case 22: if (checkquest(9224) == -1) setquest 9224; break;
	case 24: if (checkquest(13188) == -1) setquest 13188; break;
	case 25: if (isbegin_quest(12318) == 0) setquest 12318; break;
	}
	if (.@idx == 5) GeffenTournament = gettimetick(2) + 86400;
	dispbottom "Entrada autorizada. Cooldown desta instancia: 24 horas.";
	return;
