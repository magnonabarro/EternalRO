//===== rAthena Script =======================================
//= Reset NPC
//===== Description: =========================================
//= Resets skills, stats, or both.
//===== Additional Comments: =================================
//= 1.0 First Version
//= 1.1 Optimized for the greater good. [Kisuka]
//= 1.2 Cleaning [Euphy]
//= 1.3 All statuses removed upon skill reset. [Euphy]
//= 1.4 Compressed Script, Added limit use option [Stolao]
//=	Changed set -> setarray, Improved text with F_InsertPlural
//= 1.5 Added sc_end_class to reset related status changes [sader1992]
//============================================================
sala_vip,120,92,3	script	Garota Resetadora	124,{
	//	 		Skills,	Stats,	Both,	Limit
	setarray .@Reset,	1,	1,	10,	0;
	mes "[Garota Resetadora]";
	if(.@Reset[3] && reset_limit > .@Reset[3]) {
		mes "Desculpe, você só pode reiniciar "+callfunc("F_InsertPlural",.@Reset[3],"time")+" .";
		close;
	}
	mes "Eu sou a garota do Resetadora.";
	mes "Reseta Stats: "+ callfunc("F_InsertComma",.@Reset[1]) +"z";
	mes "Reseta Habilidades: "+ callfunc("F_InsertComma",.@Reset[0]) +"z";
	mes "Reseta Ambos: "+ callfunc("F_InsertComma",.@Reset[2]) +"z";
	if(.@Reset[3]) mes "Reseta "+callfunc("F_InsertPlural",.@Reset[3],"time")+", então "+((.@Reset[3]>1)?"them":"it")+" vamos?.";
	mes "Favor selecionar o serviço que deseja:";
	next;
	set .@i,(select("^FF3355Reseta Habilidades:Reseta Stat:Reseta Ambos^000000:Cancela"));
	if(.@i > 3) close;
	mes "[Garota Resetadora]";
	if (Zeny < .@Reset[.@i-1]) {
		mes "Desculpe, você não tem Zeny suficiente.";
		close;
	}
	if(.@Reset[3]){
		mes "Você só pode reiniciar "+callfunc("F_InsertPlural",.@Reset[3],"time")+" em sua vida, você está certo?";
		if(select("Deixe-me pensar: Tudo bem.") == 1) close;
	}
	set Zeny, Zeny-.@Reset[.@i-1];
	if(.@i&1){
		sc_end_class;
		ResetSkill;
	}
	if(.@i&2) ResetStatus;
	mes "Aí está!";
	if(.@Reset[3]) set reset_limit,reset_limit + 1;
	close;
	
	OnInit:
	npcshopattach "shop_hourly",1;
	waitingroom "[Resetadora]",0;
	end;
	
}
