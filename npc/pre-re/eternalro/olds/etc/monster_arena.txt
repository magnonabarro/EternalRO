function	script	duelkill	{
	if (strcharinfo(0) != $@duelist1$ && strcharinfo(0) != $@duelist2$) end;
	announce strcharinfo(0) + " ganhou o duelo!",1;
	set @fighting,0;
	set #heal,0;
	set @healing,0;
	end;
}

	announce "Morte ilegal por " + strcharinfo(0) + " detectada.",1;
	percentheal -100,-100;
	end;
}
