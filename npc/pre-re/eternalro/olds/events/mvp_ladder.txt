prontera,160,256,3	script	Portal Escada MvP	56,{  //Novo local adicionado pelo [ADM] Eragonat PK.

//prontera,164,171,3	script	Portal Escada MvP	56,{  //Local Original.
	mes "[Escada MvP]";
	mes "Diga... você quer jogar o jogo Escada MvP?";
	next;
	switch(select("Sim, vamos em frente!:Informações.:Mostre-me o melhor Record.:Não.")) {
	case 1:
		break;
	case 2:
		mes "[Escada MvP]";
		mes "Neste jogo, seu grupo tem que matar todos os monstros MvP em ordem crescente, do mais fraco ao mais forte.";
		if ( .finish_item_amount )
		mes "Se o seu grupo conseguir matar todos os MvP's e finalizar o Evento Escada MvP, cada membro ganhará "+ callfunc("F_InsertPlural", .finish_item_amount, getitemname( .finish_item_id )) +".";
		if ( .register_cost )
		mes "Mas a taxa de entrada é "+ callfunc( "F_InsertComma", .register_cost ) +" zeny(s).";
		next;
		mes "[Escada MvP]";
		mes "Você perderá o jogo se não conseguir terminar em" + .timeout + "minuto(s), ou se todo o seu grupo estiver morto.";
		mes "Boa sorte!";
		close;
	case 3:
		mes "[Escada MvP]";
		if ( !$mvpladdderparty_time ) {
			mes "Ninguém finalizou este jogo ainda.";
			close;
		}
		mes "O melhor Record é:";
		mes "[ "+( $mvpladdderparty_time / 60 )+" minuto(s) "+( $mvpladdderparty_time % 60 )+" sec ]";
		mes "Pelo Grupo ^FF0000"+ $mvpladdderparty_name$ +"^000000.";
		.@size = getarraysize( $mvpladderparty_member$ );
		for ( .@i = 0; .@i < .@size; .@i++ )
			mes "^000000"+ ( .@i +1 ) +". ^0000FF"+ $mvpladderparty_member$[.@i];
		if ( getgroupid() < .gmlvlreset ) close;
		next;
		if ( select( "Fechar.", "Redefinir Record." ) == 1 ) close;
		if ( select( "Não importa.", "Eu realmente quero reinicializá-lo.." ) == 1 ) close;
		$mvpladdderparty_time = 0;
		$mvpladdderparty_name$ = "";
		deletearray $mvpladderparty_member$[.@i];
		mes "[Escada MvP]";
		mes "Record resetado com sucesso.";
		close;
	case 4:
		mes "[Escada MvP]";
		mes "Quando você for forte o suficiente para completar o jogo, por favor volte.";
		close;
	}
	if ( !getcharid(1) ) {
		mes "[Escada MvP]";
		mes "Você tem que formar um grupo para poder jogar.";
		close;
	}
	if ( is_party_leader() == false ) {
		mes "[Escada MvP]";
		mes "Apenas o líder do grupo pode se registrar.";
		close;
	}
	.@origin = getcharid(3);
	getpartymember getcharid(1), 1;
	getpartymember getcharid(1), 2;
	for ( .@i = 0; .@i < $@partymembercount; .@i++ ) {
		if ( isloggedin( $@partymemberaid[.@i], $@partymembercid[.@i] ) ) {
			attachrid $@partymemberaid[.@i];
			if ( strcharinfo(3) == strnpcinfo(4) )
				.@online++;
		}
	}
	attachrid .@origin;
	if ( $@partymembercount != .register_min ) {
		mes "[Escada MvP]";
		mes "Você tem que formar um grupo com exatamente "+ .register_min +" membros para poder jogar.";
		close;
	}
	else if ( .@online != .register_min )  {
		mes "[Escada MvP]";
		mes "Seu grupo deverá ter "+ .register_min +" membros online no mapa '"+ strnpcinfo(4) +"'.";
		close;
	}
	else if ( .register_cost && Zeny < .register_cost ) {
		mes "[Escada MvP]";
		mes "Você não tem zeny suficiente. Por favor, volte quando tiver.";
		close;
	}
	else if ( .party_id ) {
		mes "[Escada MvP]";
		mes "Sinto muito, mas um grupo já está jogando no momento. Por favor espere até que o grupo termine.";
		mes "Obrigado.";
		close;
	}
	Zeny -= .register_cost;
	announce "O Grupo ["+ strcharinfo(1) +"] iniciou uma partida no Escada MvP.", bc_all;
	set .party_id, getcharid(1);
	set .@time_enter, gettimetick(2);
	for ( .@i = 0; .@i < $@partymembercount; .@i++ ) {
		if ( isloggedin( $@partymemberaid[.@i], $@partymembercid[.@i] ) ) {
			attachrid $@partymemberaid[.@i];
			if ( strcharinfo(3) == strnpcinfo(4) ) {
				announce "Você tem "+ .timeout +" minutos para completar "+ .totalround +" rodada(s).", bc_self;
				.@name$[.@c] = strcharinfo(0);
				.@c++;
			}
		}
	}
	cleanmap .eventmap$;
	warpparty .eventmap$, 0,0, .party_id, strnpcinfo(4);
	donpcevent strnpcinfo(0)+"::OnMvpDead";
	sleep .timeout * 60000;
	if ( .round == .totalround +1 ) {
		getpartymember .party_id, 1;
		getpartymember .party_id, 2;
		mapannounce .eventmap$, "Parabéns... Você conseguiu derrotar todos os MVPs!", bc_map;
		for ( .@i = 0; .@i < $@partymembercount; .@i++ ) {
			if ( isloggedin( $@partymemberaid[.@i], $@partymembercid[.@i] ) ) {
				attachrid $@partymemberaid[.@i];
				if ( strcharinfo(3) == .eventmap$ )
					getitem .finish_item_id, .finish_item_amount;
			}
		}
		set .@timeused, gettimetick(2) - .@time_enter;
		if ( .bonus_item_amount && .@timeused < .bonus_time * 60 ) {
			mapannounce .eventmap$, "Você será recompensado com um item bônus por completar o Escada MvP em "+ .bonus_time +" minuto(s).", bc_map;
			for ( .@i = 0; .@i < $@partymembercount; .@i++ ) {
				if ( isloggedin( $@partymemberaid[.@i], $@partymembercid[.@i] ) ) {
					attachrid $@partymemberaid[.@i];
					if ( strcharinfo(3) == .eventmap$ )
						getitem .bonus_item_id, .bonus_item_amount;
				}
			}
		}

		if ( !$mvpladdderparty_time || .@timeused < $mvpladdderparty_time ) {
			mapannounce .eventmap$, "E você quebrou o recorde! [ "+( .@timeused / 60 )+" minuto(s) "+( .@timeused % 60 )+" segundo(s) ]", bc_map;
			set $mvpladdderparty_time, .@timeused;
			set $mvpladdderparty_name$, getpartyname( .party_id );
			copyarray $mvpladderparty_member$, .@name$, .register_min;
		}
		else
			mapannounce .eventmap$, "Tempo usado [ "+( .@timeused / 60 )+" minuto(s) "+( .@timeused % 60 )+" segundo(s) ]", bc_map;
		sleep 10000;
		announce "O Grupo ["+ getpartyname( .party_id ) +"] finalizou o jogo Escada MvP!", bc_all;
	}
	else
		announce "O Grupo ["+ getpartyname( .party_id ) +"] não conseguiu finalizar o jogo Escada MvP!.", bc_all;
	mapwarp .eventmap$, .map$, .x, .y;
	killmonsterall .eventmap$;
	.party_id = .round = 0;
	end;

OnMvpDead:
	getpartymember .party_id, 1;
	getpartymember .party_id, 2;
	.round++;
	if ( .round >= 2 && .round != .totalround +1 && .round_item_amount ) {
		for ( .@i = 0; .@i < $@partymembercount; .@i++ ) {
			if ( isloggedin( $@partymemberaid[.@i], $@partymembercid[.@i] ) ) {
				attachrid $@partymemberaid[.@i];
				if ( strcharinfo(3) == .eventmap$ )
					getitem .round_item_id, .round_item_amount;
			}
		}
	}
	if ( .round == .totalround +1 ) {
		awake strnpcinfo(0);
		end;
	}
	else if ( .round == .totalround )
		mapannounce .eventmap$, "A rodada final começará em "+ .delay +" segundos...", bc_map;
	else
		mapannounce .eventmap$, "A rodada "+ .round +" começará em "+ .delay +" segundos...", bc_map;
	sleep .delay * 1000;
	if ( .mvpid[.round] == 1646 )  // Escolher aleatoriamente MVP Bio3
		.mvpid[.round] = rand(1646,1651);
	monster .eventmap$,0,0, "--já--", .mvpid[.round], 1, strnpcinfo(0)+"::OnMvpDead";
	mapannounce .eventmap$, getmonsterinfo( .mvpid[.round], MOB_NAME ) +" gerou!", bc_map|bc_blue;
	end;

OnPCLogoutEvent:
	if ( hp > 0 )
		.@less_one = 1;
	else
		end;
OnPCDieEvent:
	if ( strcharinfo(3) != .eventmap$ || !getcharid(1) ) end;
	if ( getcharid(1) != .party_id ) end;
	getpartymember .party_id, 1;
	getpartymember .party_id, 2;
	for ( .@i = 0; .@i < $@partymembercount; .@i++ ) {
		if ( isloggedin( $@partymemberaid[.@i], $@partymembercid[.@i] ) ) {
			attachrid $@partymemberaid[.@i];
			if ( strcharinfo(3) == .eventmap$ && hp > 0 )
				.@alive++;
		}
	}
	if ( .@less_one )
		.@alive--;
	if ( !.@alive ) {
		mapannounce .eventmap$, "Sem Grupos!", bc_map;
		sleep 10000;
		awake strnpcinfo(0);
	}
	end;

OnInit:
//	Configurações -----------------------------------------------------

	// Limite de Tempo (em minutos)
	// Quando o tempo acabar, todos os jogadores dentro da sala serão expulsos.
	// NÃO ajuste isso para zero!
	set .timeout, 60;

	// Taxa de Entrada (em Zeny)
	set .register_cost, 100000;

	// quantidade exata de membros do grupo necessários para iniciar o jogo
	set .register_min, 2;

	// Id de cada mvp. Você pode adicionar mais
	
	setarray .mvpid[1],
		1086,//	Besouro Ladrão 		64
		1115,//	Eddga				65
		1150,//	Flor do Luar		67
		1159,//	Freeoni		    	69

		1087,//	Orc Herói			77

		1038,//	Osíris				78
		1658,//	Egnigem Cenia		79
		1272,//	Senhor das Trevas	80

		1147,//	Maya				81

		1623,//	RSX 0806			86

	// Número de Rodadas (padrão: 39)
	set .totalround, getarraysize(.mvpid) -1;

	// Recompensa de item por completar cada rodada
	set .round_item_id, 512;
	set .round_item_amount, 1;

	// Recompensa de item por completar o jogo inteiro
	set .finish_item_id, 501;
	set .finish_item_amount, 10;

	// Recompensa bônus se o jogo for concluído dentro de um certo tempo (em minutos)
	set .bonus_time, 45; // Se completado dentro de 45 minutos, esta recompensa será dada
	set .bonus_item_id, 504;
	set .bonus_item_amount, 10;

	// Tempo entre as rodadas, em segundos (padrão: 3)
	set .delay, 5;

	// Nível mínimo de GM para redefinir o melhor Record
	set .gmlvlreset, 99;

	// Mapa do Evento
	set .eventmap$, "guild_vs2-2";

	// Configuração do mapflag
	setarray .@mapflag,
		mf_nowarp,
		mf_nowarpto,
		mf_nosave,
		mf_nomemo,
		mf_noteleport,
//		mf_nopenalty, // Desativa a perda de exp
		mf_noreturn,
//		mf_nobranch,
//		mf_nomobloot, // Desativa os pontos de queda de monstros,
//		mf_nomvploot, // 2 destes
		mf_partylock;

//	Final das Configurações --------------------------------------------------------------

	mapannounce .eventmap$, "Um administrador atualizou o servidor. Por favor, registre-se novamente. Pedimos desculpas pelo inconveniente.", bc_map;
	getmapxy .map$, .x, .y, BL_NPC;
	mapwarp .eventmap$, .map$, .x, .y;
	killmonsterall .eventmap$;
	.@size = getarraysize( .@mapflag );
	for ( .@i = 0; .@i < .@size; .@i++ )
		setmapflag .eventmap$, .@mapflag[.@i];
	end;
}
