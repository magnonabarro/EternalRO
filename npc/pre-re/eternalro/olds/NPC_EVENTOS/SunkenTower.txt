prontera,173,182,2	script	Leticia#s_alberta	4_m_soulreaper,{
    cutin "soul_van04.bmp",2;
    if (BaseLevel < 40) {
	    mes "["+strnpcinfo(1)+"]", "Você é muito fraco! Volte quando estiver mais forte.";
        close3;
    }

    // ID of cooldown quest
    .@cooldown = 12541; 

    if(checkquest(.@cooldown,PLAYTIME) == 2){
		erasequest .@cooldown;
	}
	else if (checkquest(.@cooldown,PLAYTIME) != -1){
		mes "["+strnpcinfo(1)+"]", "A distorção da Sunken Tower ainda está afetando você.", "Volte quando o efeito tiver passado.";
		close3;
	}

	.@party_id = getcharid(1);
	.@instance$ = "Sunken Tower";

	if (.@party_id < 1) {
		mes "["+strnpcinfo(1)+"]", "Hmm? Onde está seu grupo? Por favor, forme um grupo, mesmo que esteja indo sozinho.";
		close3;
	}

    if (BaseLevel >= 240) {
        .@i = 240;
        .@maxLvl = 260;
    }
    else {
        for (.@i = 40; .@i < 250; .@i+=20) {
            if (BaseLevel >= .@i && BaseLevel < .@i + 20) {
                .@maxLvl = .@i + 19;
                break;
            }
        }
    }
    
    mes "["+strnpcinfo(1)+"]", "Quer ajudar a limpar a Sunken Tower??";
    next;
	switch(select("Cancelar", "Criar " + .@instance$ + " " + .@i + "~" + .@maxLvl,"Enter " + .@instance$ + " " + .@i + "~" + .@maxLvl)){
		case 2:
			if(is_party_leader() == false){
				mes "Você não é o líder deste Grupo.";
				close3;
			}
            if(!instance_check_party(.@party_id, 1, .@i, .@maxLvl)) {
                mes "["+strnpcinfo(1)+"]", "Parece que alguém em seu grupo não é elegível para entrar nesse nível.";
                close3;
            }
            if (instance_create(.@instance$) < 0) {
                mes "["+strnpcinfo(1)+"]", "Erro ao criar a instância.";
                close3;
            }
            setinstancevar('rank, (.@i - 40) / 20, instance_id(IM_PARTY));
			close3;
		case 3:
            if (instance_id(IM_PARTY) > 0) {
                if (getinstancevar('partyCount, instance_id(IM_PARTY)) > 0) {
                    mes "["+strnpcinfo(1)+"]", "A limpeza já está em andamento. Não posso permitir que você participe mais.";
                    close3;
                }
            }
			switch(instance_enter(.@instance$)){
				case IE_OK:
					if (isbegin_quest(.@cooldown) == 0) {
						setquest .@cooldown;
						mapannounce "alberta", "" + strcharinfo(0) + " do Grupo, " + getpartyname( getcharid(1) ) + ", está entrando " + .@instance$ + ".",bc_map,"0x00FF99";
					}
					close3;
                case IE_NOMEMBER:
				case IE_OTHER:
				case IE_NOINSTANCE:
                default:
					if (is_party_leader(.@party_id) == false) {
						mes "^ff0000Observe que o Memorial Dungeon não pode ser acessado por 72 horas após sua última visita.^000000";
						next;
					}
					mes "^ff0000O líder de seu grupo ainda não criou a masmorra.^000000";
					close3;
			 }
        case 1:
            close3;
	}
    close3;

}

1@ch_u,8,55,5	script	Leticia#s_tower	4_4JOB_LETICIA,{
    cutin "4job_leticia_01.png",2;
    if (!is_party_leader()) {
        mes "["+strnpcinfo(1)+"]", "Quem é você? Não fale comigo a menos que você seja o líder.";
        close3;
    }
    'partyId = getcharid(1);

    L_MainMenu:
        if (!.hasTalked) {
            mes "["+strnpcinfo(1)+"]", "Parece que a água ainda está fluindo.", "Eu diria que há cerca de 30 minutos até que vocês estejam todos debaixo d'água.", "É melhor se apressar.";
        } else {
            mes "["+strnpcinfo(1)+"]", "O tempo está se esgotando agora. O que você vai fazer??";
        }
        mes "---------------";
        mes "Dificuldade atual: ^0000CCLevel " + 'difficulty;
        .hasTalked = 1;
        next;
        switch(select("Aguarde um momento!", "Vamos la.", "^0000CCDificuldade de mudança")) {
            case 1:
                close3;
            case 2:
                donpcevent instance_npcname("#s_tower_controller")+"::OnFirstStart";
                hideonnpc instance_npcname(strnpcinfo(0));
                close3;
            case 3:
                // You can change the item needed and required amount here
                .@itemNeeded = 35012;
                .@itemCount = 1;
                mes "["+strnpcinfo(1)+"]", "Com 10 " + mesitemlink(.@itemNeeded) + ", você pode aumentar o número e a força dos monstros.", "Não é preciso dizer que você também ganhará mais experiência.";
                mes "---------------";
                mes "Dificuldade atual: ^0000CCLevel " + 'difficulty;
                next;
                .@selection = select("Nevermind.", "Dificuldade 1", "Dificuldade 2", "Dificuldade 3", "Dificuldade 4", "Dificuldade 5");
                if (.@selection == 1)
                    close3;

                if (countitem(.@itemNeeded) < .@itemCount) {
                    mes "["+strnpcinfo(1)+"]", "O que está acontecendo? Você não tem o suficiente " + mesitemlink(.@itemNeeded) + ".";
                    close3;
                }

                if ('level == (.@selection - 1)) {
                    mes "["+strnpcinfo(1)+"]", "O que você está fazendo? A dificuldade atual já é ^0000CCLevel " + 'difficulty + ".";
                    close3;
                }

                mes "["+strnpcinfo(1)+"]", "Você deseja alterar a dificuldade para ^0000CCLevel " + (.@selection - 1) + "?";
                next;
                if (select("Nevermind.", "Sim, mude-o.") == 1) {
                    close3;
                }

                'difficulty = .@selection - 1;
                mapannounce 'map$, "Leticia: A dificuldade agora é de nível " + 'difficulty + ".", bc_map,0xCC6666,FW_NORMAL,14;
                delitem .@itemNeeded, .@itemCount;
                close3;
        }

    close3;
    end;
}
